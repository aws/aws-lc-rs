12/// Joins a literal flag prefix (e.g. `-I`, `-L`, `-Wl,-rpath,`) with a path
13/// without forcing UTF-8.
14fn prefixed_os_arg(prefix: &str, value: &OsStr) -> std::ffi::OsString {
15    let mut s = std::ffi::OsString::from(prefix);
16    s.push(value);
17    s
18}
19
20/// Runs `program` with owned `OsString` args.
21fn execute_os(program: &OsStr, args: &[std::ffi::OsString]) -> crate::CommandResult {
22    let refs: Vec<&OsStr> = args.iter().map(std::ffi::OsString::as_os_str).collect();
23    execute_command(program, &refs)
24}

26/// Extra libraries the FIPS link probe must name when linking a *static*
27/// libcrypto. The probe is bare C and so lacks the std-provided pthread/dl/c
28/// linkage a real Rust binary would have. Each entry must be linkable on the
29/// target (Apple has no standalone `-ldl`) and is a no-op where folded into
30/// libc (e.g. glibc >= 2.34).
31fn fips_probe_static_deps() -> &'static [&'static str] {
32    match target_os().as_str() {
33        "linux" | "android" => &["-pthread", "-ldl", "-lm"],
34        // *BSD: threads via `-pthread`; dl/m symbols live in libc.
35        "freebsd" | "openbsd" | "netbsd" | "dragonfly" => &["-pthread"],
36        // Apple (libSystem) and anything else (incl. windows-gnu): nothing
37        // portable to add; a genuine missing dep still surfaces as a link
38        // error whose diagnostic names transitive dependencies.
39        _ => &[],
40    }
41}

43/// Returns cc-rs's target compiler flags, minus LTO flags.
44fn inherited_probe_args(compiler: &cc::Tool) -> Vec<std::ffi::OsString> {
45    compiler
46        .args()
47        .iter()
48        .filter(|arg| !arg.to_str().is_some_and(is_lto_flag))
49        .cloned()
50        .collect()
51}

53/// Appends the probe source and link arguments to the inherited compiler flags.
54fn append_probe_link_args(
55    args: &mut Vec<std::ffi::OsString>,
56    is_cl_like: bool,
57    include_dir: &Path,
58    crypto_lib: &ResolvedLib,
59    lib_dir: &Path,
60    probe_src: &Path,
61    exec_path: &Path,
62) {
63    if is_cl_like {
64        let obj_path = out_dir().join("aws_lc_fips_link_probe.obj");
65        args.extend([
66            prefixed_os_arg("/I", include_dir.as_os_str()),
67            probe_src.as_os_str().to_os_string(),
68            prefixed_os_arg("/Fe:", exec_path.as_os_str()),
69            prefixed_os_arg("/Fo:", obj_path.as_os_str()),
70            "/link".into(),
71            crypto_lib.path.as_os_str().to_os_string(),
72        ]);
73    } else {
74        args.extend([
75            prefixed_os_arg("-I", include_dir.as_os_str()),
76            probe_src.as_os_str().to_os_string(),
77            prefixed_os_arg("-L", lib_dir.as_os_str()),
78            prefixed_os_arg("-Wl,-rpath,", lib_dir.as_os_str()),
79        ]);
80        if matches!(crypto_lib.lib_type, OutputLibType::Static) {
81            args.push(crypto_lib.path.as_os_str().to_os_string());
82            args.extend(fips_probe_static_deps().iter().copied().map(Into::into));
83        } else {
84            args.push(format!("-l{}", crypto_lib.name).into());
85        }
86        args.extend(["-o".into(), exec_path.as_os_str().to_os_string()]);
87    }
88}

90/// Compiles and links the FIPS probe against the install's `libcrypto`,
91/// returning the linked executable's path on success.
92pub(crate) fn compile_fips_probe(
93    manifest_dir: &Path,
94    include_dir: &Path,
95    crypto_lib: &ResolvedLib,
96    lib_dir: &Path,
97) -> Result<PathBuf, String> {
98    let probe_src = manifest_dir.join("builder").join("fips_link_probe.c");
99    if !probe_src.is_file() {
100        return Err(format!(
101            "FIPS probe source missing: {}",
102            probe_src.display()
103        ));
104    }
105    println!("cargo:rerun-if-changed={}", probe_src.display());
106
107    let exec_path = out_dir().join(if target_os() == "windows" {
108        "aws_lc_fips_link_probe.exe"
109    } else {
110        "aws_lc_fips_link_probe"
111    });
112
113    let cc_build = cc::Build::new();
114    let compiler = cc_build.get_compiler();
115    // Keyed on the compiler family (not the target ABI); unsupported families error below.
116    if !(compiler.is_like_clang() || compiler.is_like_gnu() || compiler.is_like_msvc()) {
117        return Err(format!(
118            "FIPS verification requires a Clang-, GCC-, or MSVC-compatible compiler; \
119             {} is not supported. Set CC to a supported compiler.",
120            compiler.path().display(),
121        ));
122    }
123
124    let mut args = inherited_probe_args(&compiler);
125    append_probe_link_args(
126        &mut args,
127        // Probe link-flag syntax follows the compiler driver mode, not the ABI.
128        compiler_is_cl_like(&compiler),
129        include_dir,
130        crypto_lib,
131        lib_dir,
132        &probe_src,
133        &exec_path,
134    );
135
136    let result = execute_os(compiler.path().as_os_str(), &args);
137    if !result.executed {
138        return Err(format!(
139            "FIPS verification failed: could not execute compiler {compiler}.\n\
140             Spawn error: {spawn_error}",
141            compiler = compiler.path().display(),
142            spawn_error = result.spawn_error.as_deref().unwrap_or("unknown error"),
143        ));
144    }
145    if !result.status {
146        return Err(format!(
147            "FIPS verification failed: linker rejected the probe against {lib}.\n\
148             The library is likely not an AWS-LC FIPS build (BORINGSSL_integrity_test \
149             is exported only by FIPS, non-ASAN builds); when cross-compiling it can \
150             also indicate an architecture or CRT model (/MD vs /MT) mismatch.\n\
151             Stderr: {stderr}",
152            lib = crypto_lib.path.display(),
153            stderr = result.stderr,
154        ));
155    }
156
157    Ok(exec_path)
158}

160/// Verifies that the supplied system `libcrypto` is a usable AWS-LC FIPS
161/// library by linking the FIPS-only probe and, when possible, running it.
162pub(crate) fn verify_fips_install(
163    manifest_dir: &Path,
164    include_dir: &Path,
165    crypto_lib: &ResolvedLib,
166    lib_dir: &Path,
167) -> Result<(), String> {
168    let probe = compile_fips_probe(manifest_dir, include_dir, crypto_lib, lib_dir)?;
169    run_fips_probe(&probe, crypto_lib)
170}

172const FIPS_MODE_OFF_EXIT_CODE: i32 = 42;
173
174/// Runs the probe when the host can launch a target binary.
175pub(crate) fn run_fips_probe(exec_path: &Path, crypto_lib: &ResolvedLib) -> Result<(), String> {
176    let invocation: Option<(std::ffi::OsString, Vec<std::ffi::OsString>)> = if !is_cross_compiling()
177    {
178        Some((exec_path.as_os_str().to_os_string(), Vec::new()))
179    } else if let Some(mut runner) = target_runner() {
180        // runner = [program, args...]; append the probe path last.
181        let program = runner.remove(0);
182        let mut run_args: Vec<std::ffi::OsString> = runner.into_iter().map(Into::into).collect();
183        run_args.push(exec_path.as_os_str().to_os_string());
184        Some((program.into(), run_args))
185    } else {
186        None
187    };
188
189    let Some((program, run_args)) = invocation else {
190        let _ = std::fs::remove_file(exec_path);
191        emit_warning(
192            "Cross-compile detected with no target runner: skipping runtime FIPS_mode() check. \
193             Build-time link probe passed; the runtime constructor in fips_runtime_check.c \
194             will assert FIPS mode at process startup.",
195        );
196        return Ok(());
197    };
198
199    let runtime_result = execute_os(program.as_os_str(), &run_args);
200    let _ = std::fs::remove_file(exec_path);
201
202    if !runtime_result.executed {
203        emit_warning(format!(
204            "FIPS runtime check skipped: probe linked but could not be executed via {program}. \
205             Relying on the build-time link probe and the runtime constructor. \
206             Spawn error: {spawn_error}. Stderr: {stderr}",
207            program = program.to_string_lossy(),
208            spawn_error = runtime_result
209                .spawn_error
210                .as_deref()
211                .unwrap_or("unknown error"),
212            stderr = runtime_result.stderr,
213        ));
214        return Ok(());
215    }
216    if runtime_result.status {
217        emit_warning(
218            "FIPS verification: build-time link probe and runtime FIPS_mode() check passed.",
219        );
220        return Ok(());
221    }
222    if runtime_result.exit_code == Some(FIPS_MODE_OFF_EXIT_CODE) {
223        return Err(format!(
224            "FIPS verification failed: probe linked but FIPS_mode() returned 0 at \
225             runtime against {lib}. The library exposes FIPS-only symbols but is \
226             not running in FIPS mode (e.g. a FIPS+ASAN build, which the AWS-LC \
227             FIPS module disables at runtime).\n\
228             Stderr: {stderr}",
229            lib = crypto_lib.path.display(),
230            stderr = runtime_result.stderr,
231        ));
232    }
233    emit_warning(format!(
234        "FIPS runtime check skipped: probe ran via {program} but exited unexpectedly. \
235         Relying on the build-time link probe and the runtime constructor. \
236         Exit code: {exit_code:?}. Stderr: {stderr}",
237        program = program.to_string_lossy(),
238        exit_code = runtime_result.exit_code,
239        stderr = runtime_result.stderr,
240    ));
241
242    Ok(())
243}

245/// Returns `CARGO_TARGET_<TRIPLE>_RUNNER` as `[program, args...]`, if set.
246fn target_runner() -> Option<Vec<String>> {
247    let key = format!(
248        "CARGO_TARGET_{}_RUNNER",
249        target().replace(['-', '.'], "_").to_uppercase()
250    );
251    println!("cargo:rerun-if-env-changed={key}");
252    let value = std::env::var(&key).ok()?;
253    match parse_shell_words(&value) {
254        Ok(parts) if parts.is_empty() => None,
255        Ok(parts) => Some(parts),
256        Err(err) => {
257            emit_warning(format!("Ignoring invalid {key}: {err}"));
258            None
259        }
260    }
261}
262
263fn parse_shell_words(value: &str) -> Result<Vec<String>, String> {
264    #[derive(Clone, Copy)]
265    enum Quote {
266        Single,
267        Double,
268    }
269
270    let mut parts = Vec::new();
271    let mut current = String::new();
272    let mut quote = None;
273    let mut escape = false;
274
275    for ch in value.chars() {
276        if escape {
277            current.push(ch);
278            escape = false;
279            continue;
280        }
281
282        match quote {
283            Some(Quote::Single) => {
284                if ch == '\'' {
285                    quote = None;
286                } else {
287                    current.push(ch);
288                }
289            }
290            Some(Quote::Double) => match ch {
291                '"' => quote = None,
292                '\\' => escape = true,
293                _ => current.push(ch),
294            },
295            None => match ch {
296                '\'' => quote = Some(Quote::Single),
297                '"' => quote = Some(Quote::Double),
298                '\\' => escape = true,
299                _ if ch.is_whitespace() => {
300                    if !current.is_empty() {
301                        parts.push(std::mem::take(&mut current));
302                    }
303                }
304                _ => current.push(ch),
305            },
306        }
307    }
308
309    if escape {
310        return Err("runner ends with an unfinished escape".to_string());
311    }
312    if quote.is_some() {
313        return Err("runner contains an unmatched quote".to_string());
314    }
315    if !current.is_empty() {
316        parts.push(current);
317    }
318
319    Ok(parts)
320}
