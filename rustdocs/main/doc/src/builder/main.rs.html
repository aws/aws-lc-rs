90impl EnvGuard {
91    fn new<T: AsRef<OsStr>>(key: &str, new_value: T) -> Self {
92        let original_value = env::var(key).ok();
93        env::set_var(key, new_value);
94        Self {
95            key: key.to_string(),
96            original_value,
97        }
98    }
99
100    fn remove(key: &str) -> Self {
101        let original_value = env::var(key).ok();
102        env::remove_var(key);
103        Self {
104            key: key.to_string(),
105            original_value,
106        }
107    }
108}
109
110impl Drop for EnvGuard {
111    fn drop(&mut self) {
112        if let Some(ref value) = self.original_value {
113            env::set_var(&self.key, value);
114        } else {
115            env::remove_var(&self.key);
116        }
117    }
118}

197// "EFFECTIVE_TARGET" => "SYS_EFFECTIVE_TARGET_aarch64_unknown_linux_gnu" + "SYS_EFFECTIVE_TARGET"
198fn optional_env_crate_target<N: AsRef<str>>(name: N) -> Option<String> {
199    let name = name.as_ref();
200    let crate_name = crate_name().to_uppercase().replace('-', "_");
201    let name_for_crate = format!("{crate_name}_{name}");
202    let name_for_crate_target = env_name_for_target(&name_for_crate);
203    optional_env(name_for_crate_target).or(optional_env(name_for_crate))
204}

206pub(crate) fn optional_env_target<N: AsRef<str>>(name: N) -> Option<String> {
207    let name_for_target = env_name_for_target(name.as_ref());
208    optional_env(name_for_target).or(optional_env(name))
209}

211fn optional_env<N: AsRef<str>>(name: N) -> Option<String> {
212    let name = name.as_ref();
213    println!("cargo:rerun-if-env-changed={name}");
214    if let Ok(value) = env::var(name) {
215        emit_warning(format!("Environment Variable found '{name}': '{value}'"));
216        return Some(value);
217    }
218    None
219}

267fn parse_to_bool(env_var_value: &str) -> Option<bool> {
268    let env_var_value = env_var_value.to_lowercase();
269    if env_var_value.starts_with('0')
270        || env_var_value.starts_with('n')
271        || env_var_value.starts_with("off")
272        || env_var_value.starts_with('f')
273    {
274        emit_warning(format!("Value: {env_var_value} is false."));
275        return Some(false);
276    }
277    if env_var_value.starts_with(|c: char| c.is_ascii_digit())
278        || env_var_value.starts_with('y')
279        || env_var_value.starts_with("on")
280        || env_var_value.starts_with('t')
281    {
282        emit_warning(format!("Value: {env_var_value} is true."));
283        return Some(true);
284    }
285    None
286}

288impl Default for OutputLibType {
289    fn default() -> Self {
290        if let Some(stc_lib) = is_static_library() {
291            if stc_lib {
292                OutputLibType::Static
293            } else {
294                OutputLibType::Dynamic
295            }
296        } else if !is_fips_build()
297            || (is_fips_build()
298                && (target_os() == "linux" || target_os().ends_with("bsd"))
299                && (target_arch() == "x86_64" || target_arch() == "aarch64"))
300        {
301            OutputLibType::Static
302        } else {
303            OutputLibType::Dynamic
304        }
305    }
306}

308impl OutputLibType {
309    fn rust_lib_type(&self) -> &str {
310        match self {
311            OutputLibType::Static => "static",
312            OutputLibType::Dynamic => "dylib",
313        }
314    }
315
316    /// Returns the `kind` portion of a `cargo:rustc-link-lib=<kind>=<name>`
317    /// directive, including the `+whole-archive` modifier when whole-archive
318    /// linking has been requested for a static library.
319    ///
320    /// Whole-archive linking forces the linker to include every object file
321    /// from the static archive, which surfaces unresolved symbol references
322    /// inside object files that no consumer happens to call. We use it in CI
323    /// to catch missing assembly source files in the `cc_builder` source lists.
324    fn rust_link_lib_kind(self) -> String {
325        if matches!(self, OutputLibType::Static) && is_link_whole_archive() {
326            format!("{}:+whole-archive", self.rust_lib_type())
327        } else {
328            self.rust_lib_type().to_string()
329        }
330    }
331
332    fn opposite(self) -> Self {
333        match self {
334            Self::Static => Self::Dynamic,
335            Self::Dynamic => Self::Static,
336        }
337    }
338
339    fn description(self) -> &'static str {
340        match self {
341            Self::Static => "static",
342            Self::Dynamic => "dynamic",
343        }
344    }
345}

421const MAX_CMD_OUTPUT_SIZE: usize = 1 << 15;
422fn execute_command(executable: &OsStr, args: &[&OsStr]) -> CommandResult {
423    match Command::new(executable).args(args).output() {
424        Ok(mut result) => {
425            let exit_code = result.status.code();
426            result.stderr.truncate(MAX_CMD_OUTPUT_SIZE);
427            let stderr = String::from_utf8(result.stderr)
428                .unwrap_or_default()
429                .into_boxed_str();
430            result.stdout.truncate(MAX_CMD_OUTPUT_SIZE);
431            let stdout = String::from_utf8(result.stdout)
432                .unwrap_or_default()
433                .into_boxed_str();
434            CommandResult {
435                stderr,
436                stdout,
437                spawn_error: None,
438                exit_code,
439                executed: true,
440                status: result.status.success(),
441            }
442        }
443        Err(err) => CommandResult {
444            stderr: String::new().into_boxed_str(),
445            stdout: String::new().into_boxed_str(),
446            spawn_error: Some(err.to_string().into_boxed_str()),
447            exit_code: None,
448            executed: false,
449            status: false,
450        },
451    }
452}

511/// Whether the target ABI is MSVC (a `*-pc-windows-msvc` target).
512///
513/// This reflects the target ABI only and is the right signal for ABI/CRT/SDK
514/// decisions (MSVC CRT linkage, `_CRT_SECURE_NO_WARNINGS`, the `clang-cl`
515/// dependency checks). It is *not* a reliable signal for the compiler's flag
516/// *dialect*: a `*-pc-windows-msvc` target can be built with plain `clang` in
517/// GNU driver mode (e.g. `CC=clang`), which needs GNU-style flags. For dialect
518/// decisions use `compiler_is_cl_like` instead.
519fn target_is_msvc() -> bool {
520    target_env() == "msvc"
521}
522
523/// Whether the active C compiler is driven in MSVC "cl" mode (`cl.exe` or
524/// `clang-cl`), which is what determines cl-style flag *dialect* (`/Fo`, `/FI`,
525/// `/Od`, no `--param ...`).
526///
527/// Driver mode is a property of the *compiler*, not the target ABI, and the two
528/// can diverge in both directions:
529/// - plain `clang` targeting `*-pc-windows-msvc` runs in GNU mode and needs
530///   GNU-style flags (so the target ABI alone is not enough), and
531/// - `cc::Tool::is_like_msvc()` can misclassify `clang-cl` as plain Clang when
532///   its probe misfires in a sandbox (e.g. cargo-xwin), emitting GNU-only flags
533///   to a cl-mode driver (so `is_like_msvc()` alone is not enough).
534///
535/// `clang` selects its driver mode from `argv[0]`, so a `clang-cl`/`cl` program
536/// name is an authoritative fallback when `cc`'s family probe is unreliable.
537/// See: <https://github.com/aws/aws-lc-rs/issues/1146>
538pub(crate) fn compiler_is_cl_like(compiler: &cc::Tool) -> bool {
539    compiler.is_like_msvc() || program_name_is_cl_driver(compiler.path())
540}
541
542/// Whether a compiler program name selects clang's cl driver mode (`clang-cl`)
543/// or is `cl` itself. Robust fallback for `compiler_is_cl_like`.
544pub(crate) fn program_name_is_cl_driver(path: &Path) -> bool {
545    match path.file_name().and_then(OsStr::to_str) {
546        Some(name) => {
547            let name = name.to_ascii_lowercase();
548            name.contains("clang-cl") || name == "cl" || name == "cl.exe"
549        }
550        None => false,
551    }
552}

606/// On Windows, convert a path to its 8.3 short form to avoid MAX_PATH (260 char) limits
607/// when cl.exe is invoked with deeply nested source trees (e.g. Bazel runfiles).
608#[cfg(windows)]
609fn to_short_path(path: &Path) -> PathBuf {
610    use std::os::windows::ffi::{OsStrExt, OsStringExt};
611    extern "system" {
612        fn GetShortPathNameW(
613            lpszLongPath: *const u16,
614            lpszShortPath: *mut u16,
615            cchBuffer: u32,
616        ) -> u32;
617    }
618    let wide: Vec<u16> = path
619        .as_os_str()
620        .encode_wide()
621        .chain(std::iter::once(0))
622        .collect();
623    let len = unsafe { GetShortPathNameW(wide.as_ptr(), std::ptr::null_mut(), 0) };
624    if len == 0 {
625        return path.to_path_buf();
626    }
627    let mut buf = vec![0u16; len as usize];
628    let result = unsafe { GetShortPathNameW(wide.as_ptr(), buf.as_mut_ptr(), len) };
629    if result == 0 {
630        return path.to_path_buf();
631    }
632    buf.truncate(result as usize);
633    let short_path = PathBuf::from(std::ffi::OsString::from_wide(&buf));
634
635    const MAX_PATH: usize = 260;
636    let original_len = wide.len() - 1;
637    if original_len >= MAX_PATH && (result as usize) >= MAX_PATH {
638        emit_warning(format!(
639            "Path length ({}) exceeds MAX_PATH ({}) and 8.3 short name conversion was ineffective. \
640             8.3 short names may be disabled on this volume. \
641             See: https://learn.microsoft.com/en-us/windows-server/administration/windows-commands/fsutil-8dot3name",

653fn get_builder(prefix: &Option<String>, manifest_dir: &Path, out_dir: &Path) -> Box<dyn Builder> {
654    let cmake_builder_builder = || {
655        Box::new(CmakeBuilder::new(
656            manifest_dir.to_path_buf(),
657            out_dir.to_path_buf(),
658            prefix.clone(),
659            OutputLibType::default(),
660        ))
661    };
662
663    if let Some(install_dir) = get_system_dir_path() {
664        // SYSTEM_DIR and USE_SYSTEM=0 are contradictory ("use this install" vs
665        // "never use a system install"); fail rather than silently honoring one.
666        assert!(
667            use_system() != Some(false),
668            "{system_dir} is set but {use_system}=0 forbids a system install.\n\
669             Unset one: keep {system_dir} to link it, or {use_system}=0 to build from source.",
670            system_dir = crate_env_var_name("SYSTEM_DIR"),
671            use_system = crate_env_var_name("USE_SYSTEM"),
672        );
673        return Box::new(SystemLib::new(
674            manifest_dir.to_path_buf(),
675            install_dir,
676            get_system_bindings_path(),
677            get_system_skip_version_check(),
678        ));
679    }
680
681    // No explicit SYSTEM_DIR. Unless the user opted out with USE_SYSTEM=0, try
682    // to auto-detect a usable system AWS-LC. Detection is non-fatal: an
683    // unsuitable (or absent) install falls through to the source build, unless
684    // USE_SYSTEM=1 demands one.
685    if use_system() != Some(false) {
686        if let Some(system_lib) = system_detect::detect_system_awslc(manifest_dir) {
687            // Mirror the explicit path's global state so is_bindgen_required()
688            // and the other get_system_dir_path() consumers agree we are
689            // linking a system library.
690            set_system_dir(system_lib.marker_dir().to_path_buf());
691            return Box::new(system_lib);
692        }
693        // Detection found nothing usable. Fall through to a source build unless
694        // USE_SYSTEM=1 demands a system install, in which case fail.
695        assert!(
696            use_system() != Some(true),
697            "{use_system}=1 requires a system-installed AWS-LC, but none was found.\n\
698             Set {system_dir} to point at an install, provide OPENSSL_DIR, or unset \
699             {use_system} to build from source.\n\
700             Re-run with `-vv` to see why each discovered candidate was rejected.",
701            use_system = crate_env_var_name("USE_SYSTEM"),
702            system_dir = crate_env_var_name("SYSTEM_DIR"),
703        );
704    }
705
706    if is_fips_build() {
707        return cmake_builder_builder();
708    }
709
710    let cc_builder_builder = || {
711        Box::new(CcBuilder::new(
712            manifest_dir.to_path_buf(),
713            out_dir.to_path_buf(),
714            prefix.clone(),
715            OutputLibType::default(),
716        ))
717    };
718
719    if let Some(val) = is_cmake_builder() {
720        let builder: Box<dyn Builder> = if val {
721            cmake_builder_builder()
722        } else {
723            cc_builder_builder()
724        };
725        builder.check_dependencies().unwrap();
726        return builder;
727    } else if is_no_asm() || sanitizer().is_some() {
728        let builder = cmake_builder_builder();
729        builder.check_dependencies().unwrap();
730        return builder;
731    } else if !is_bindgen_required() {
732        let cc_builder = cc_builder_builder();
733        if cc_builder.check_dependencies().is_ok() {
734            return cc_builder;
735        }
736    }
737    let cmake_builder = cmake_builder_builder();
738    cmake_builder.check_dependencies().unwrap();
739    cmake_builder
740}

1021/// Returns true if jitter entropy should be built. This is false when the user
1022/// explicitly set `AWS_LC_SYS_NO_JITTER_ENTROPY=1`, or when auto-detection
1023/// determined that required headers are unavailable.
1024fn should_build_jitter_entropy() -> bool {
1025    static AVAILABLE: OnceLock<bool> = OnceLock::new();
1026
1027    // If the user explicitly set the env var, respect it unconditionally.
1028    if let Some(val) = disable_jitter_entropy() {
1029        return !val;
1030    }
1031

1033        // wasm/emscripten targets do not support CPU jitter entropy.
1034        if target_arch().starts_with("wasm") {
1035            return false;
1036        }
1037        // The current FIPS branch does not include jitter entropy.
1038        if is_fips_build() {
1039            return true;
1040        }
1041        // Only Apple targets need the CoreServices header.
1042        if target_vendor() != "apple" {
1043            return true;
1044        }
1045        probe_apple_core_services()
1046    })

1049/// Probes whether `CoreServices/CoreServices.h` is available for the current
1050/// target toolchain. On Apple targets, jitterentropy requires this header which
1051/// is only present when the macOS SDK is installed. When cross-compiling to
1052/// Apple targets from non-macOS hosts (e.g., Linux → macOS via cargo-zigbuild),
1053/// this header is typically missing.
1054fn probe_apple_core_services() -> bool {
1055    let probe_dir = out_dir().join("out-apple_core_services_check");
1056    let src_dir = probe_dir.join("src");
1057    let _ = std::fs::create_dir_all(&src_dir);
1058    let source_file = src_dir.join("apple_core_services_check.c");
1059    if std::fs::write(&source_file, "#include <CoreServices/CoreServices.h>\n").is_err() {
1060        emit_warning("Failed to write CoreServices probe file; assuming available.");
1061        return true;
1062    }
1063
1064    let mut cc_build = cc::Build::default();
1065    cc_build.file(&source_file).out_dir(&probe_dir);
1066    let available = cc_build.try_compile_intermediates().is_ok();
1067    let _ = std::fs::remove_dir_all(&probe_dir);
1068
1069    if !available {
1070        emit_warning(
1071            "CoreServices/CoreServices.h not available; \
1072             automatically disabling CPU jitter entropy.",
1073        );
1074    }
1075    available
1076}

1164fn find_clang_cl_in_vs() -> Option<OsString> {
1165    // Use the cc crate's VS discovery (which calls vswhere internally) to
1166    // locate the VS installation, then look for clang-cl relative to it.
1167    let tool = cc::windows_registry::find_tool(&target(), "cl.exe")?;
1168    let cl_path = tool.path().to_path_buf();
1169
1170    // cl.exe lives deep inside the VS installation:
1171    //   <VS>/VC/Tools/MSVC/<ver>/bin/<host>/<target>/cl.exe
1172    // clang-cl lives at:
1173    //   <VS>/VC/Tools/Llvm/<arch>/bin/clang-cl.exe
1174    // Walk up from cl.exe until we find the VS root.
1175    for ancestor in cl_path.ancestors() {
1176        let vc_llvm = ancestor.join("VC").join("Tools").join("Llvm");
1177        if vc_llvm.exists() {
1178            for arch_dir in &["ARM64", "x64"] {
1179                let clang_cl = vc_llvm.join(arch_dir).join("bin").join("clang-cl.exe");
1180                if clang_cl.exists()
1181                    && execute_command(clang_cl.as_os_str(), &["--version".as_ref()]).status
1182                {
1183                    emit_warning(format!("Found clang-cl at: {}", clang_cl.display()));
1184                    return Some(clang_cl.into_os_string());
1185                }
1186            }
1187            break;
1188        }
1189    }
1190
1191    None
1192}

1256/// Links the startup FIPS-mode self-check for the system-library FIPS path.
1257pub(crate) fn link_fips_runtime_check(
1258    manifest_dir: &Path,
1259    include_dir: &Path,
1260) -> Result<(), String> {
1261    let source = manifest_dir.join("builder").join("fips_runtime_check.c");
1262    if !source.is_file() {
1263        return Err(format!(
1264            "FIPS runtime check source missing: {}",
1265            source.display()
1266        ));
1267    }
1268    println!("cargo:rerun-if-changed={}", source.display());
1269
1270    let mut cc_build = cc::Build::default();
1271    cc_build
1272        .file(&source)
1273        .warnings(false)
1274        // We emit the link directives ourselves (with +whole-archive), so
1275        // suppress cc-rs's automatic emission.
1276        .cargo_metadata(false)
1277        // Silence cc-rs's `ar cqD` probe, which Apple's `ar` rejects (and
1278        // cc-rs leaks as warnings before retrying). Real compile errors still
1279        // surface from the build-time probe in verify_fips_install().
1280        .cargo_warnings(false)
1281        .include(include_dir);
1282
1283    let lib_name = "aws_lc_fips_runtime_check";
1284    cc_build.compile(lib_name);
1285
1286    // Keep the constructor object in the final link.
1287    println!("cargo:rustc-link-search=native={}", out_dir().display());
1288    println!("cargo:rustc-link-lib=static:+whole-archive={lib_name}");
1289    Ok(())
1290}

1395/// Emits the shared post-build cargo metadata for source-based builders
1396/// (`CMake` and CC). This sets up include paths, exports library and
1397/// configuration names for downstream crates, and registers rerun triggers.
1398pub(crate) fn emit_source_build_metadata(manifest_dir: &Path) {
1399    // Only aws-lc-fips-sys consumes the generated FIPS-version constant; a
1400    // FIPS-flavored aws-lc-sys build reports a module version of 0.
1401    if is_fips_crate() {
1402        system_library::emit_fips_version(&get_aws_lc_include_path(manifest_dir)).unwrap();
1403    }
1404
1405    // MinGW/GCC ignores `#pragma comment(lib, "bcrypt.lib")`, so we must
1406    // link explicitly. The upstream CMakeLists.txt forces _WIN32_WINNT_WIN7
1407    // for MINGW+GCC, activating the BCryptGenRandom codepath.
1408    // See: https://github.com/aws/aws-lc/pull/3239
1409    if target().contains("-windows-gnu") {
1410        println!("cargo:rustc-link-lib=bcrypt");
1411    }
1412
1413    println!(
1414        "cargo:include={}",
1415        setup_include_paths(&out_dir(), manifest_dir).display()
1416    );
1417
1418    // export the artifact names
1419    println!("cargo:libcrypto={}_crypto", prefix_string());
1420    if cfg!(feature = "ssl") {
1421        println!("cargo:libssl={}_ssl", prefix_string());
1422    }
1423
1424    println!("cargo:rerun-if-changed=aws-lc/");
1425}

1427fn setup_include_paths(out_dir: &Path, manifest_dir: &Path) -> PathBuf {
1428    let mut include_paths = vec![
1429        get_rust_include_path(manifest_dir),
1430        get_generated_include_path(manifest_dir),
1431        get_aws_lc_include_path(manifest_dir),
1432    ];
1433
1434    if let Some(extra_paths) = get_env_includes_path() {
1435        include_paths.extend(extra_paths);
1436    }
1437
1438    let include_dir = out_dir.join("include");
1439    std::fs::create_dir_all(&include_dir).unwrap();
1440
1441    // iterate over all the include paths and copy them into the final output
1442    for path in include_paths {
1443        for child in std::fs::read_dir(path).into_iter().flatten().flatten() {
1444            if child.path().is_file() {
1445                std::fs::copy(
1446                    child.path(),
1447                    include_dir.join(child.path().file_name().unwrap()),
1448                )
1449                .expect("Failed to copy include file during build setup");
1450                continue;
1451            }
1452
1453            // prefer the earliest paths
1454            let options = fs_extra::dir::CopyOptions::new()
1455                .skip_exist(true)
1456                .copy_inside(true);
1457            fs_extra::dir::copy(child.path(), &include_dir, &options)
1458                .expect("Failed to copy include directory during build setup");
1459        }
1460    }
1461
1462    include_dir
1463}

1483fn verify_bindgen() -> Result<(), String> {
1484    if !is_external_bindgen_requested().unwrap_or(true) {
1485        return Err("external bindgen usage prevented by SYS_EXTERNAL_BINDGEN=0".to_string());
1486    }
1487
1488    let result = execute_command("bindgen".as_ref(), &["--version".as_ref()]);
1489    if !result.status {
1490        if result.executed {
1491            eprintln!(
1492                "bindgen-cli exited with an error status:\nSTDOUT: {}\n\nSTDERR: {}",
1493                result.stdout, result.stderr
1494            );
1495        } else {
1496            eprintln!(
1497                "Consider installing the bindgen-cli: \
1498            `cargo install --force --locked bindgen-cli`\
1499            \n\
1500            See our User Guide for more information about bindgen:\
1501            https://aws.github.io/aws-lc-rs/index.html"
1502            );
1503        }
1504        return Err("External bindgen command failed.".to_string());
1505    }
1506    let mut major_version: u32 = 0;
1507    let mut minor_version: u32 = 0;
1508    let mut patch_version: u32 = 0;
1509    let bindgen_version = result.stdout.split(' ').nth(1);
1510    if let Some(version) = bindgen_version {
1511        let version_parts: Vec<&str> = version.trim().split('.').collect();
1512        if version_parts.len() == 3 {
1513            major_version = version_parts[0].parse::<u32>().unwrap_or(0);
1514            minor_version = version_parts[1].parse::<u32>().unwrap_or(0);
1515            patch_version = version_parts[2].parse::<u32>().unwrap_or(0);
1516        }
1517    }
1518    // We currently expect to support all bindgen versions >= 0.69.5
1519    if major_version == 0 && (minor_version < 69 || (minor_version == 69 && patch_version < 5)) {
1520        eprintln!(
1521            "bindgen-cli was used. Detected version was: \
1522            {major_version}.{minor_version}.{patch_version} \n\
1523        Consider upgrading : \
1524        `cargo install --force --locked bindgen-cli`\
1525        \n\
1526        See our User Guide for more information about bindgen:\
1527        https://aws.github.io/aws-lc-rs/index.html"
1528        );
1529    }
1530    Ok(())
1531}

1561fn invoke_external_bindgen(
1562    manifest_dir: &Path,
1563    prefix: &Option<String>,
1564    gen_bindings_path: &Path,
1565) -> Result<(), String> {
1566    verify_bindgen()?;
1567
1568    emit_warning(format!(
1569        "Generating bindings - external bindgen. Platform: {}",
1570        effective_target()
1571    ));
1572
1573    let _guard_target = EnvGuard::new("TARGET", effective_target());
1574
1575    let options = BindingOptions {
1576        // We collect the symbols w/o the prefix added
1577        build_prefix: None,
1578        include_ssl: false,
1579        disable_prelude: true,
1580    };
1581
1582    let clang_args = prepare_clang_args(manifest_dir, &options);
1583    let header = get_rust_include_path(manifest_dir)
1584        .join("rust_wrapper.h")
1585        .display()
1586        .to_string();
1587
1588    let sym_prefix: String;
1589    let mut bindgen_params = vec![];
1590    if let Some(prefix_str) = prefix {
1591        sym_prefix = if target_os().to_lowercase() == "macos"
1592            || target_os().to_lowercase() == "ios"
1593            || target_os().to_lowercase() == "tvos"
1594            || (target_os().to_lowercase() == "windows" && target_arch() == "x86")
1595        {
1596            format!("_{prefix_str}_")
1597        } else {
1598            format!("{prefix_str}_")
1599        };
1600        bindgen_params.extend(vec!["--prefix-link-name", sym_prefix.as_str()]);
1601    }
1602
1603    // These flags needs to be kept in sync with the setup in bindgen::prepare_bindings_builder
1604    // If `bindgen-cli` makes backwards incompatible changes, we will update the parameters below
1605    // to conform with the most recent release. We will guide consumers to likewise use the
1606    // latest version of bindgen-cli.
1607    bindgen_params.extend(vec![
1608        "--allowlist-file",
1609        r".*(/|\\)openssl((/|\\)[^/\\]+)+\.h",
1610        "--allowlist-file",
1611        r".*(/|\\)rust_wrapper\.h",
1612        "--rustified-enum",
1613        r"point_conversion_form_t",
1614        "--default-macro-constant-type",
1615        r"signed",
1616        "--with-derive-default",
1617        "--with-derive-partialeq",
1618        "--with-derive-eq",
1619        "--raw-line",
1620        COPYRIGHT,
1621        "--generate",
1622        "functions,types,vars,methods,constructors,destructors",
1623        header.as_str(),
1624        "--rust-target",
1625        r"1.70",
1626        "--output",
1627        gen_bindings_path.to_str().unwrap(),
1628        "--formatter",
1629        r"rustfmt",
1630    ]);
1631    if !options.disable_prelude {
1632        bindgen_params.extend(["--raw-line", PRELUDE]);
1633    }
1634    bindgen_params.push("--");
1635
1636    for x in &clang_args {
1637        bindgen_params.push(x.as_str());
1638    }
1639    let cmd_params: Vec<OsString> = bindgen_params.iter().map(OsString::from).collect();
1640    let cmd_params: Vec<&OsStr> = cmd_params.iter().map(OsString::as_os_str).collect();
1641
1642    let result = execute_command("bindgen".as_ref(), cmd_params.as_ref());
1643    if !result.status {
1644        return Err(format!(
1645            "\n\n\
1646            bindgen-PARAMS: {}\n\
1647            bindgen-STDOUT: {}\n\
1648            bindgen-STDERR: {}",
1649            bindgen_params.join(" "),
1650            result.stdout.as_ref(),
1651            result.stderr.as_ref()
1652        ));
1653    }
1654    Ok(())
1655}
