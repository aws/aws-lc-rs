4//! Support for linking against a system-installed AWS-LC library.
5//!
6//! Activated when `<crate>_SYSTEM_DIR` (e.g. `AWS_LC_SYS_SYSTEM_DIR`) is set.
7//! Two entry points are invoked from the build script: `check_dependencies`
8//! locates and resolves the libraries, and `link()` then validates the
9//! include directory, checks the installed version against the bundled
10//! one, locates the bindings, copies them into `OUT_DIR`, and emits the
11//! appropriate `cargo:` directives.
12

15    crate_env_var_name, emit_rustc_cfg, emit_warning, is_fips_build, is_fips_crate,

17    OutputLibType,
18};

22/// Resolved on-disk locations for a system AWS-LC.
23///
24/// Prefix-based discovery (explicit `*_SYSTEM_DIR`, `OPENSSL_DIR`, default
25/// prefixes) derives every field from a single install prefix via
26/// [`InstallLayout::from_prefix`]. Sources that yield *independent* locations
27/// — pkg-config, the `OPENSSL_INCLUDE_DIR`/`OPENSSL_LIB_DIR` split, multiarch
28/// library directories such as `/usr/lib/<triple>` — populate the fields
29/// directly via [`InstallLayout::from_paths`].
30pub(crate) struct InstallLayout {
31    /// Directory passed to the compiler with `-I`; contains `openssl/base.h`.
32    include_dir: PathBuf,
33    /// Library search directories, highest priority first.
34    lib_dirs: Vec<PathBuf>,
35    /// Prefix under which conventional bindings live
36    /// (`<prefix>/share/rust/aws_lc_bindings.rs`), when discovery could
37    /// determine one. `None` means only an explicit `SYSTEM_BINDINGS` override
38    /// can supply bindings.
39    bindings_prefix: Option<PathBuf>,
40    /// Representative directory used in diagnostics and recorded as the active
41    /// system install once adopted: the install prefix when known, otherwise
42    /// the include directory.
43    marker_dir: PathBuf,
44}
45
46impl InstallLayout {
47    /// Layout for a conventional single-prefix install: headers at
48    /// `<prefix>/include`, libraries under `<prefix>/lib64` (preferred on
49    /// 64-bit targets when present) then `<prefix>/lib`, bindings under
50    /// `<prefix>/share/rust`.
51    pub(crate) fn from_prefix(prefix: PathBuf) -> Self {
52        let include_dir = prefix.join("include");
53        let mut lib_dirs = Vec::new();
54        // Cargo always sets CARGO_CFG_TARGET_POINTER_WIDTH for build scripts.
55        let target_is_64bit =
56            std::env::var("CARGO_CFG_TARGET_POINTER_WIDTH").is_ok_and(|w| w == "64");
57        let lib64 = prefix.join("lib64");
58        if target_is_64bit && lib64.exists() {
59            lib_dirs.push(lib64);
60        }
61        lib_dirs.push(prefix.join("lib"));
62        Self {
63            include_dir,
64            lib_dirs,
65            bindings_prefix: Some(prefix.clone()),
66            marker_dir: prefix,
67        }
68    }
69
70    /// Layout for independently-located headers and libraries. `bindings_prefix`
71    /// is where `share/rust/aws_lc_bindings.rs` is expected (e.g. derived from a
72    /// pkg-config include path or the `OPENSSL_INCLUDE_DIR` install layout).
73    pub(crate) fn from_paths(
74        include_dir: PathBuf,
75        lib_dirs: Vec<PathBuf>,
76        bindings_prefix: Option<PathBuf>,
77    ) -> Self {
78        let marker_dir = bindings_prefix
79            .clone()
80            .unwrap_or_else(|| include_dir.clone());
81        Self {
82            include_dir,
83            lib_dirs,
84            bindings_prefix,
85            marker_dir,
86        }
87    }
88}

90pub(crate) struct SystemLib {
91    pub manifest_dir: PathBuf,
92    layout: InstallLayout,
93    pub bindings_file: Option<PathBuf>,
94    pub skip_version_check: bool,
95    pub crypto_lib: RefCell<Option<ResolvedLib>>,
96    pub ssl_lib: RefCell<Option<ResolvedLib>>,
97    /// Memoizes the bindings path resolved by `validate`, which doubles as a
98    /// "validation already succeeded" marker. Auto-detection probes an install
99    /// (running `validate`) and, on success, links the *same* instance; this
100    /// cache prevents the subsequent `link` from repeating the version check,
101    /// bindings lookup, and — most importantly — re-compiling and re-running
102    /// the FIPS probe binary.
103    validated_bindings: RefCell<Option<PathBuf>>,
104    /// Memoizes the FIPS module version resolved by `validate`, so `link`
105    /// need not re-parse `base.h`. `None` until resolved (or when
106    /// `skip_version_check` bypassed resolution).
107    fips_version: RefCell<Option<u32>>,
108}

165impl Builder for SystemLib {
166    fn check_dependencies(&self) -> Result<(), String> {
167        let requested_lib_type = is_static_library().map(|stc| {
168            if stc {
169                OutputLibType::Static
170            } else {
171                OutputLibType::Dynamic
172            }
173        });
174        // Resolve crypto and (optionally) ssl in the same lib_dir, so we don't
175        // commit to a lib_dir for crypto only to discover ssl is missing there
176        // when it lives in the alternate dir.
177        //
178        // Track the most-informative error we encounter so that when no
179        // candidate lib_dir works we surface the actionable resolver error
180        let mut last_err: Option<String> = None;
181        for lib_dir in &self.layout.lib_dirs {
182            let crypto_lib = match resolve_library(
183                lib_dir.as_path(),
184                requested_lib_type,
185                "crypto",
186                CRYPTO_LIB_CANDIDATES,
187            ) {
188                Ok(lib) => lib,
189                Err(e) => {
190                    last_err = Some(e);
191                    continue;
192                }
193            };
194            if cfg!(feature = "ssl") {
195                let ssl_lib = match resolve_library(
196                    lib_dir.as_path(),
197                    requested_lib_type,
198                    "ssl",
199                    SSL_LIB_CANDIDATES,
200                ) {
201                    Ok(lib) => lib,
202                    Err(e) => {
203                        last_err = Some(e);
204                        continue;
205                    }
206                };
207                self.ssl_lib.replace(Some(ssl_lib));
208            }
209            self.crypto_lib.replace(Some(crypto_lib));
210            return Ok(());
211        }
212        Err(last_err.unwrap_or_else(|| {
213            format!(
214                "AWS-LC libcrypto{ssl} not found under {dirs}",
215                ssl = if cfg!(feature = "ssl") { "/libssl" } else { "" },
216                dirs = self
217                    .layout
218                    .lib_dirs
219                    .iter()
220                    .map(|d| d.display().to_string())
221                    .collect::<Vec<_>>()
222                    .join(", "),
223            )
224        }))
225    }
226
227    fn build(&self) -> Result<(), String> {
228        self.link()
229    }
230
231    fn name(&self) -> &'static str {
232        "System Library"
233    }
234}

236impl SystemLib {
237    /// Performs every *non-emitting* check required before linking against the
238    /// resolved install, returning the path to the pre-generated bindings to
239    /// use. In order, it: confirms the include directory exists, verifies the
240    /// AWS-LC marker and enforces the version / FIPS-version floor (unless
241    /// skipped), locates the bindings, and — for FIPS builds — verifies the
242    /// FIPS module.
243    ///
244    /// This is the single source of truth for "is this install suitable?".
245    /// Both the fatal explicit path (via `link`) and the non-fatal
246    /// auto-detection path (via `probe`) route through here, so the two can
247    /// never diverge — a probe that succeeds guarantees the subsequent link
248    /// will not fail validation. The result is memoized so a probe followed by
249    /// a link does not repeat the work (notably the FIPS probe).
250    ///
251    /// `check_dependencies` must have run first to resolve the libraries.
252    fn validate(&self) -> Result<PathBuf, String> {
253        if let Some(bindings) = self.validated_bindings.borrow().clone() {
254            return Ok(bindings);
255        }
256
257        let crypto_lib = self.crypto_lib.borrow();
258        let crypto_lib = crypto_lib
259            .as_ref()
260            .expect("check_dependencies must run first");
261        let lib_dir = crypto_lib
262            .path
263            .parent()
264            .ok_or("libcrypto parent directory not found")?;
265        let include_dir = &self.layout.include_dir;
266        if !include_dir.exists() {
267            return Err(format!(
268                "Include directory not found: {}\n\
269                 Verify the AWS-LC headers exist (via {}, OPENSSL_DIR, or \
270                 OPENSSL_INCLUDE_DIR).",
271                include_dir.display(),
272                crate_env_var_name("SYSTEM_DIR"),
273            ));
274        }
275
276        if self.skip_version_check {
277            emit_warning("Skipping AWS-LC version compatibility check.");
278            read_and_validate_base_h(include_dir)?;
279        } else if is_fips_build() {
280            // FIPS gates on the FIPS module version (see MINIMUM_FIPS_VERSION),
281            // not the library version string.
282            let fips_version = validate_and_resolve_fips_version(include_dir)?;
283            self.fips_version.replace(Some(fips_version));
284            if fips_version < MINIMUM_FIPS_VERSION {
285                return Err(format!(
286                    "AWS-LC FIPS module version too old: installed FIPS version {fips_version} < \
287                     minimum supported {MINIMUM_FIPS_VERSION}.\n\
288                     Please use a newer AWS-LC FIPS build or unset {} to build from source.\n\
289                     To bypass this check (not recommended), set {}=1",
290                    crate_env_var_name("SYSTEM_DIR"),
291                    crate_env_var_name("SYSTEM_SKIP_VERSION_CHECK"),
292                ));
293            }
294        } else {
295            let version = validate_and_extract_version(include_dir)?;
296            if !version_at_least(&version, MINIMUM_AWS_LC_VERSION)? {
297                return Err(format!(
298                    "AWS-LC version too old: installed {version} < minimum supported \
299                     {MINIMUM_AWS_LC_VERSION}.\n\
300                     Please upgrade AWS-LC or unset {} to build from source.\n\
301                     To bypass this check (not recommended), set {}=1",
302                    crate_env_var_name("SYSTEM_DIR"),
303                    crate_env_var_name("SYSTEM_SKIP_VERSION_CHECK"),
304                ));
305            }
306        }
307
308        let bindings =
309            resolve_bindings(self.layout.bindings_prefix.as_deref(), &self.bindings_file)?;
310
311        if is_fips_build() {
312            // Verify the supplied FIPS library is a usable FIPS module before
313            // committing to it. The matching startup self-check link directive
314            // is emitted later, in `link`.
315            verify_fips_install(
316                self.manifest_dir.as_path(),
317                include_dir,
318                crypto_lib,
319                lib_dir,
320            )?;
321        }
322
323        self.validated_bindings.replace(Some(bindings.clone()));
324        Ok(bindings)
325    }
326
327    /// Non-fatally checks whether this install is suitable to link against. It
328    /// resolves the libraries and runs the exact same validation `link` relies
329    /// on, but emits no `cargo:` directives. Auto-detection treats any `Err`
330    /// as "not suitable — fall back to a source build" rather than a hard
331    /// error.
332    pub(crate) fn probe(&self) -> Result<(), String> {
333        self.check_dependencies()?;
334        self.validate().map(|_| ())
335    }
336
337    /// Emits the `cargo:` directives needed to link against the
338    /// system-installed AWS-LC.
339    pub(crate) fn link(&self) -> Result<(), String> {
340        let bindings = self.validate()?;
341
342        let crypto_lib = self.crypto_lib.borrow();
343        let crypto_lib = crypto_lib
344            .as_ref()
345            .expect("check_dependencies must run first");
346        let lib_dir = crypto_lib
347            .path
348            .parent()
349            .ok_or("libcrypto parent directory not found")?;
350        let include_dir = &self.layout.include_dir;
351
352        std::fs::copy(&bindings, out_dir().join("bindings.rs"))
353            .map_err(|e| format!("Failed to copy bindings from {}: {}", bindings.display(), e))?;
354        emit_warning(format!(
355            "Using pre-generated bindings from: {}",
356            bindings.display()
357        ));
358        emit_rustc_cfg("use_bindgen_pregenerated");
359
360        emit_warning(format!(
361            "Using system-installed AWS-LC from: {}",
362            self.layout.marker_dir.display()
363        ));
364
365        if is_fips_crate() {
366            // Only aws-lc-fips-sys consumes the generated constant; a
367            // FIPS aws-lc-sys build reports 0.
368            let fips_version = if let Some(version) = *self.fips_version.borrow() {
369                version
370            } else {
371                emit_warning(
372                    "AWS-LC version check skipped; aws_lc_rs::fips_version() will report None.",
373                );
374                0
375            };
376            write_fips_version(fips_version)?;
377        }
378        if is_fips_build() {
379            // The FIPS module itself was already verified in `validate`; here
380            // we only add the startup self-check.
381            link_fips_runtime_check(self.manifest_dir.as_path(), include_dir)?;
382        }
383
384        self.emit_link_directives(crypto_lib, include_dir, lib_dir);
385        Ok(())
386    }
387
388    fn emit_link_directives(&self, crypto_lib: &ResolvedLib, include_dir: &Path, lib_dir: &Path) {
389        let optional_ssl_lib = self.ssl_lib.borrow();
390        let kind = crypto_lib.lib_type.rust_lib_type();
391        println!("cargo:rustc-link-search=native={}", lib_dir.display());
392        println!("cargo:libdir={}", lib_dir.display());
393        println!("cargo:libcrypto={}", crypto_lib.name);
394        println!("cargo:libcrypto_path={}", crypto_lib.path.display());
395        println!("cargo:link_kind={kind}");
396        println!("cargo:rustc-link-lib={kind}={}", crypto_lib.name);
397        if let Some(ssl_lib) = optional_ssl_lib.as_ref() {
398            // `link_kind` (above) describes libcrypto. Without a requested lib
399            // type the two libraries could in principle resolve to different
400            // types; warn rather than export misleading metadata.
401            if ssl_lib.lib_type != crypto_lib.lib_type {
402                emit_warning(format!(
403                    "libcrypto ({}) and libssl ({}) resolved to different library types; \
404                     the exported link_kind describes libcrypto only",
405                    crypto_lib.lib_type.description(),
406                    ssl_lib.lib_type.description(),
407                ));
408            }
409            println!("cargo:rustc-link-lib={kind}={}", ssl_lib.name);
410            println!("cargo:libssl={}", ssl_lib.name);
411            println!("cargo:libssl_path={}", ssl_lib.path.display());
412            println!("cargo:rerun-if-changed={}", ssl_lib.path.display());
413        }
414
415        // After the libcrypto/libssl directives: GNU ld needs the providers to
416        // follow the archive that references them.
417        crate::emit_system_libs_metadata();
418
419        println!("cargo:include={}", include_dir.display());
420
421        println!("cargo:rerun-if-changed={}", include_dir.display());
422        println!("cargo:rerun-if-changed={}", crypto_lib.path.display());
423    }
424}

430/// Minimum AWS-LC (mainline) version supported via the system-library path,
431/// for the non-FIPS `aws-lc-sys` crate. `1.68.0` is the first release that can
432/// emit Rust bindings (`-DGENERATE_RUST_BINDINGS=ON`), which the
433/// system-library path consumes. A declared floor (not derived from the
434/// bundled submodule) whose support is proven by CI; bump it and the CI pin
435/// together. See `.github/workflows/system-lib-tests.yml`.
436const MINIMUM_AWS_LC_VERSION: &str = "1.68.0";
437
438/// Minimum AWS-LC FIPS *module* version supported via the system-library path,
439/// for `aws-lc-fips-sys`. The FIPS version (aws/aws-lc#3211) is decoupled from
440/// the library version and increases monotonically across FIPS branches, so it
441/// is a stable basis for comparison.
442///
443/// FIPS 4.x (`fips-2025-09-12-lts`) is the floor: aws-lc-rs unconditionally
444/// references symbols (e.g. ML-DSA/PQDSA) that older FIPS modules do not
445/// provide.
446const MINIMUM_FIPS_VERSION: u32 = 4;
447
448/// Finds the first `#define <name> ...` line in `base.h` content and returns
449/// the value token following the macro name (or `""` for a bare `#define
450/// <name>` with no value), or `None` if the macro is not `#define`d. Matching
451/// on the leading whitespace-separated tokens — rather than a bare `contains`
452/// — prevents a comment that merely mentions the macro name from
453/// false-matching.
454fn find_define<'a>(content: &'a str, name: &str) -> Option<&'a str> {
455    content.lines().find_map(|line| {
456        let mut tokens = line.split_whitespace();
457        if tokens.next() == Some("#define") && tokens.next() == Some(name) {
458            // Map a valueless `#define <name>` to "" so callers still
459            // distinguish "defined" (`Some`) from "absent" (`None`).
460            Some(tokens.next().unwrap_or(""))
461        } else {
462            None
463        }
464    })
465}

467/// Reads `<include_dir>/openssl/base.h`, verifies the `OPENSSL_IS_AWSLC` marker
468/// (rejecting `OpenSSL`/`BoringSSL`), and returns the header path and contents.
469fn read_and_validate_base_h(include_dir: &Path) -> Result<(PathBuf, String), String> {
470    let base_h = include_dir.join("openssl").join("base.h");
471    let content = std::fs::read_to_string(&base_h)
472        .map_err(|e| format!("Failed to read {}: {}", base_h.display(), e))?;
473
474    let has_awslc_marker = find_define(&content, "OPENSSL_IS_AWSLC").is_some();
475    if !has_awslc_marker {
476        return Err(format!(
477            "Headers at {} are not valid AWS-LC headers.\n\
478             The OPENSSL_IS_AWSLC marker was not found in base.h.\n\
479             Ensure the path contains AWS-LC headers, not OpenSSL or BoringSSL.",
480            include_dir.display()
481        ));
482    }
483
484    Ok((base_h, content))
485}

487/// Validates the headers are AWS-LC and returns the library version string
488/// (`AWSLC_VERSION_NUMBER_STRING`).
489fn validate_and_extract_version(include_dir: &Path) -> Result<String, String> {
490    let (base_h, content) = read_and_validate_base_h(include_dir)?;
491    extract_version(&base_h, &content)
492}

494/// Validates the headers are AWS-LC and returns the FIPS module version.
495fn validate_and_resolve_fips_version(include_dir: &Path) -> Result<u32, String> {
496    let (base_h, content) = read_and_validate_base_h(include_dir)?;
497    resolve_fips_version(&base_h, &content)

500/// Resolves the FIPS module version from `base.h` and generates the constant
501/// consumed by `aws-lc-fips-sys::fips_version()`. Used by the source-build
502/// path; the system-library path reuses the version resolved by `validate`.
503pub(crate) fn emit_fips_version(include_dir: &Path) -> Result<(), String> {
504    let (base_h, content) = read_and_validate_base_h(include_dir)?;
505    let fips_version = resolve_fips_version(&base_h, &content)?;
506    write_fips_version(fips_version)
507}

509/// Writes `OUT_DIR/fips_version.rs`: a bare `u32` literal that `include!`
510/// expands in const context, keeping `fips_version()` a `const fn`.
511pub(crate) fn write_fips_version(fips_version: u32) -> Result<(), String> {
512    let dest = out_dir().join("fips_version.rs");
513    std::fs::write(&dest, format!("{fips_version}\n"))
514        .map_err(|e| format!("Failed to write {}: {e}", dest.display()))
515}

517/// Extracts `AWSLC_VERSION_NUMBER_STRING` from already-loaded `base.h`
518/// content. Kept separate from `validate_and_extract_version` so the
519/// bundled-version guard test can call it directly, skipping the
520/// `OPENSSL_IS_AWSLC` marker check (the bundled headers are AWS-LC by
521/// construction).
522fn extract_version(base_h: &Path, content: &str) -> Result<String, String> {
523    let Some(value) = find_define(content, "AWSLC_VERSION_NUMBER_STRING") else {
524        return Err(format!(
525            "Could not find AWSLC_VERSION_NUMBER_STRING in {}.",
526            base_h.display()
527        ));
528    };
529    let trimmed = value.trim_matches('"');
530    if trimmed != value && !trimmed.is_empty() {
531        Ok(trimmed.to_string())
532    } else {
533        Err(format!(
534            "Malformed AWSLC_VERSION_NUMBER_STRING in {}",
535            base_h.display()
536        ))
537    }
538}

540fn version_at_least(installed: &str, required: &str) -> Result<bool, String> {
541    let parse = |s: &str| -> Result<(u32, u32, u32), String> {
542        let parts: Vec<&str> = s.split('.').collect();
543        if parts.len() != 3 {
544            return Err(format!("Invalid version format: {s}"));
545        }
546        Ok((
547            parts[0]
548                .parse()
549                .map_err(|_| format!("Invalid major: {}", parts[0]))?,
550            parts[1]
551                .parse()
552                .map_err(|_| format!("Invalid minor: {}", parts[1]))?,
553            parts[2]
554                .parse()
555                .map_err(|_| format!("Invalid patch: {}", parts[2]))?,
556        ))
557    };
558    Ok(parse(installed)? >= parse(required)?)
559}

561/// Resolves the FIPS module version from `base.h`: prefers the
562/// `AWSLC_FIPS_VERSION_NUMBER` macro (FIPS 4.x+ / mainline, aws/aws-lc#3211),
563/// falling back to the major of `AWSLC_VERSION_NUMBER_STRING` on legacy FIPS
564/// branches (<= 3.x) that predate the macro.
565fn resolve_fips_version(base_h: &Path, content: &str) -> Result<u32, String> {
566    if let Some(version) = extract_fips_version_number(content)? {
567        return Ok(version);
568    }
569    let version = extract_version(base_h, content)?;
570    version_major(&version)
571}
572
573/// Extracts the integer `AWSLC_FIPS_VERSION_NUMBER` macro.
574///
575/// Returns `Ok(None)` when the macro is absent (a legacy FIPS branch that
576/// predates it; callers fall back to the version-string major). A macro that is
577/// *present but unparseable* is a hard error rather than a silent fallback:
578/// post-decoupling (aws/aws-lc#3211) the library version is not a reliable
579/// substitute, so masking a malformed authoritative value would be wrong.
580fn extract_fips_version_number(content: &str) -> Result<Option<u32>, String> {
581    let Some(value) = find_define(content, "AWSLC_FIPS_VERSION_NUMBER") else {
582        return Ok(None);
583    };
584    value.parse::<u32>().map(Some).map_err(|_| {
585        let shown = if value.is_empty() { "<empty>" } else { value };
586        format!("Malformed AWSLC_FIPS_VERSION_NUMBER: expected an unsigned integer, found {shown}")
587    })
588}

590/// Parses the MAJOR component of a `MAJOR.MINOR.PATCH` version string.
591fn version_major(version: &str) -> Result<u32, String> {
592    version
593        .split('.')
594        .next()
595        .and_then(|major| major.parse::<u32>().ok())
596        .ok_or_else(|| format!("Invalid version format: {version}"))
597}

603/// Candidate bare library names for libcrypto, in preference order.
604///
605/// `crypto` covers plain AWS-LC installs (the common case). `crypto-awslc`
606/// covers installs produced with `ENABLE_DIST_PKG=ON` or a non-Apple Unix
607/// shared build that disables `ENABLE_PRE_SONAME_BUILD`, where AWS-LC's
608/// `CMake` appends the `-awslc` SONAME suffix to the library file name.
609///
610/// Note: the `BORINGSSL_PREFIX` symbol-prefixing feature does *not* rename
611/// the library file in upstream AWS-LC's build — it only renames the
612/// symbols inside. Do not try to apply the symbol prefix here.
613const CRYPTO_LIB_CANDIDATES: &[&str] = &["crypto", "crypto-awslc"];
614
615/// Candidate bare library names for libssl, in preference order.
616/// See `CRYPTO_LIB_CANDIDATES` for the rationale.
617const SSL_LIB_CANDIDATES: &[&str] = &["ssl", "ssl-awslc"];
618
619/// Resolves a library, choosing the linkage form based on `requested_lib_type`.
620///
621/// `lib_kind` is the human-readable name used in error messages
622/// (e.g. `"crypto"` or `"ssl"`).
623///
624/// When `requested_lib_type` is `Some`, the user has set `AWS_LC_SYS_STATIC`
625/// explicitly and the requested form is required: if only the other form
626/// is present we return an error rather than silently falling back. When
627/// `requested_lib_type` is `None`, we prefer the form picked by
628/// `OutputLibType::default()` but fall back to whichever form is available.
629fn resolve_library(
630    lib_dir: &Path,
631    requested_lib_type: Option<OutputLibType>,
632    lib_kind: &str,
633    candidate_names: &[&str],
634) -> Result<ResolvedLib, String> {
635    let preferred = requested_lib_type.unwrap_or_default();
636
637    if let Some(found) = find_candidate(lib_dir, preferred, candidate_names) {
638        return Ok(found);
639    }
640
641    // Preferred form not present. Probe the opposite form so we can give
642    // an accurate diagnostic: "wrong form on disk" vs "nothing on disk".
643    let opposite = preferred.opposite();
644    let opposite_match = find_candidate(lib_dir, opposite, candidate_names);
645
646    if requested_lib_type.is_some() {
647        // User explicitly requested a form that isn't available; do not
648        // silently fall back.
649        if opposite_match.is_some() {
650            return Err(format!(
651                "{var} is set to request {requested} linking, but only a {got} {lib_kind} library was found in {}.\n\
652                 Expected one of: {}\n\
653                 Provide the requested form at this location, or unset {var} to allow \
654                 the available form.",
655                lib_dir.display(),
656                expected_lib_filenames(candidate_names, Some(preferred)),
657                requested = preferred.description(),
658                got = opposite.description(),
659                var = crate_env_var_name("STATIC"),
660            ));
661        }
662        return Err(format!(
663            "{var} is set to request {requested} linking, but no {lib_kind} library was found in {}.\n\
664             Expected one of: {}",
665            lib_dir.display(),
666            expected_lib_filenames(candidate_names, Some(preferred)),
667            requested = preferred.description(),
668            var = crate_env_var_name("STATIC"),
669        ));
670    }
671
672    if let Some(found) = opposite_match {
673        let desc = opposite.description();
674        emit_warning(format!(
675            "Only {desc} {lib_kind} library found in system install; using {desc} linking."
676        ));
677        return Ok(found);
678    }
679
680    Err(format!(
681        "No {lib_kind} library found in {}\n\
682         Expected one of: {}",
683        lib_dir.display(),
684        expected_lib_filenames(candidate_names, None),
685    ))
686}
687
688/// Returns the first candidate in `candidates` whose library of the given
689/// `lib_type` exists in `lib_dir`.
690fn find_candidate(
691    lib_dir: &Path,
692    lib_type: OutputLibType,
693    candidates: &[&str],
694) -> Option<ResolvedLib> {
695    candidates.iter().find_map(|name| {
696        probe_lib(lib_dir, name, lib_type).map(|path| ResolvedLib {
697            lib_type,
698            name: (*name).to_string(),
699            path,
700        })
701    })
702}

704/// Returns `true` when the target is Windows with the MSVC toolchain.
705fn is_msvc() -> bool {
706    matches!(
707        (target_os().as_str(), target_env().as_str()),
708        ("windows", "msvc")
709    )
710}
711
712/// Probes for a library named `name` of the requested `lib_type` in `lib_dir`.
713///
714/// On MSVC, both static archives and DLL import libraries are named
715/// `{name}.lib`. The two are disambiguated by `msvc_has_sibling_dll`: if
716/// `../bin/{name}.dll` exists, the `.lib` is an import library (dynamic);
717/// otherwise it is a real static archive.
718///
719/// On MinGW, the runtime DLL is installed to `bin/` rather than `lib_dir`, so
720/// the import library `lib{name}.dll.a` is the only artifact to look for here.
721fn probe_lib(lib_dir: &Path, name: &str, lib_type: OutputLibType) -> Option<PathBuf> {
722    let want_dynamic = matches!(lib_type, OutputLibType::Dynamic);
723    let path = lib_dir.join(lib_type.library_filename(name));
724    if !path.exists() {
725        return None;
726    }
727    // MSVC: `{name}.lib` could be either a real static archive or a DLL
728    // import library. Classify by sibling-DLL presence.
729    if is_msvc() && msvc_has_sibling_dll(lib_dir, name) != want_dynamic {
730        return None;
731    }
732    Some(path)
733}
734
735/// Comma-separated list of platform-appropriate filenames the resolver looks
736/// for. Used purely for error messages. On MSVC `library_filename` returns the
737/// same `{base}.lib` for either linkage, so `filter` doesn't change the
738/// resulting list there (deduplicated below).
739fn expected_lib_filenames(candidates: &[&str], filter: Option<OutputLibType>) -> String {
740    let lib_types: &[OutputLibType] = match filter {
741        Some(OutputLibType::Static) => &[OutputLibType::Static],
742        Some(OutputLibType::Dynamic) => &[OutputLibType::Dynamic],
743        None => &[OutputLibType::Static, OutputLibType::Dynamic],
744    };
745    let mut names: Vec<String> = Vec::new();
746    for &lt in lib_types {
747        for base in candidates {
748            let name = lt.library_filename(base);
749            if !names.contains(&name) {
750                names.push(name);
751            }
752        }
753    }
754    names.join(", ")
755}
756
757/// Returns `true` when a sibling `../bin/{name}.dll` exists relative to
758/// `lib_dir`. `CMake` installs the runtime DLL under `bin/` while the import
759/// library goes to `lib/` on MSVC, so this is the canonical way to detect
760/// that the `.lib` in `lib_dir` is an import library rather than a real
761/// static archive.
762fn msvc_has_sibling_dll(lib_dir: &Path, name: &str) -> bool {
763    lib_dir
764        .parent()
765        .is_some_and(|root| root.join("bin").join(format!("{name}.dll")).exists())
766}

772/// Locates pre-generated bindings, either from the explicit override or the
773/// conventional install location populated by AWS-LC's `CMake` install (see
774/// <https://github.com/aws/aws-lc/pull/2999>, AWS-LC v1.68.0+).
775///
776/// `bindings_prefix` is the install prefix under which `share/rust/` is
777/// searched; it is `None` for discovery sources that couldn't determine a
778/// prefix (in which case only an explicit override can supply bindings).
779///
780/// A misconfigured override is a hard error rather than a silent fall-through,
781/// since the user clearly asked for those specific bindings.
782fn resolve_bindings(
783    bindings_prefix: Option<&Path>,
784    bindings_override: &Option<PathBuf>,
785) -> Result<PathBuf, String> {
786    if let Some(path) = bindings_override {
787        if path.is_file() {
788            return Ok(path.clone());
789        }
790        return Err(format!(
791            "{} does not point to a file: {}",
792            crate_env_var_name("SYSTEM_BINDINGS"),
793            path.display()
794        ));
795    }
796
797    let Some(prefix) = bindings_prefix else {
798        return Err(format!(
799            "No pre-generated bindings found for system-installed AWS-LC.\n\
800             The detected install did not expose an install prefix from which to\n\
801             locate share/rust/aws_lc_bindings.rs. Set {} to point at a bindings file.",
802            crate_env_var_name("SYSTEM_BINDINGS"),
803        ));
804    };
805
806    let conventional = prefix.join("share").join("rust").join("aws_lc_bindings.rs");
807    if conventional.is_file() {
808        return Ok(conventional);
809    }
810
811    Err(format!(
812        "No pre-generated bindings found for system-installed AWS-LC.\n\n\
813         To resolve this, either:\n\
814         1. Install AWS-LC with Rust bindings (share/rust/aws_lc_bindings.rs) under {}, or\n\
815         2. Set {} to point to a bindings file",
816        prefix.display(),
817        crate_env_var_name("SYSTEM_BINDINGS"),
818    ))
819}
