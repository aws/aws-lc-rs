21/// Strip LTO-related flags from a CFLAGS string. LTO flags interfere with the
22/// FIPS delocator pipeline, which compiles C to assembly (`-S`) and processes it
23/// with a Go tool. The combination of `-S` and `-flto` causes GCC to produce
24/// GIMPLE-annotated output that the delocator cannot handle. Additionally, LTO
25/// object files mixed with hand-written assembly in static libraries can cause
26/// linker errors (e.g. `R_X86_64_32` relocations incompatible with PIE).
27/// See also: aws-lc `CMakeLists.txt` which disables assembly when LTO is active.
28fn strip_lto_flags(cflags: &str) -> String {
29    cflags
30        .split_whitespace()
31        .filter(|flag| !is_lto_flag(flag))
32        .collect::<Vec<_>>()
33        .join(" ")
34}

43fn test_prebuilt_nasm_script(script_path: &Path) -> bool {
44    // Call with no args - both scripts will exit with error, but we only care if they can execute
45    execute_command(script_path.as_os_str(), &[]).executed

48fn find_cmake_command() -> Option<OsString> {
49    if let Some(cmake) = optional_env_optional_crate_target("CMAKE") {
50        emit_warning(format!("CMAKE environment variable set: {}", cmake.clone()));
51        if execute_command(cmake.as_ref(), &["--version".as_ref()]).status {
52            Some(cmake.into())
53        } else {
54            None
55        }
56    } else if execute_command("cmake3".as_ref(), &["--version".as_ref()]).status {
57        Some("cmake3".into())
58    } else if execute_command("cmake".as_ref(), &["--version".as_ref()]).status {
59        Some("cmake".into())
60    } else {
61        None
62    }
63}

88    fn apply_universal_build_options<'a>(
89        &self,
90        cmake_cfg: &'a mut cmake::Config,
91    ) -> &'a cmake::Config {
92        // Use the compiler options identified by CcBuilder
93        let cc_builder = CcBuilder::new(
94            self.manifest_dir.clone(),
95            self.out_dir.clone(),
96            self.build_prefix.clone(),
97            self.output_lib_type,
98        );
99        let cc_build = cc::Build::new();
100        let (is_cl_like, build_options) =
101            cc_builder.collect_universal_build_options(&cc_build, true);
102        for option in &build_options {
103            option.apply_cmake(cmake_cfg, is_cl_like);
104        }
105
106        // CMake seeds `CMAKE_C_FLAGS` and `CMAKE_ASM_FLAGS` separately, so the
107        // C-side prefix-map does not cover `.S` sources. In release-style
108        // builds, mirror it with `asmflag()` when the compiler accepts
109        // `-Wa,--debug-prefix-map=...`, and skip paths with spaces because the
110        // flag must survive CMake and the assembler as one bare token.
111        let opt_level = cargo_env("OPT_LEVEL");
112        if !is_cl_like
113            && (target_os() == "linux" || target_os().ends_with("bsd"))
114            && !matches!(opt_level.as_str(), "0" | "1" | "2")
115            && !self.manifest_dir.to_string_lossy().contains(' ')
116        {
117            let path_str = self.manifest_dir.display().to_string();
118            let asm_flag = format!("-Wa,--debug-prefix-map={path_str}=");
119            if cc_build.is_flag_supported(&asm_flag).unwrap_or(false) {
120                cmake_cfg.asmflag(asm_flag);
121            }
122        }
123
124        cmake_cfg
125    }

129    #[allow(clippy::too_many_lines)]
130    fn prepare_cmake_build(&self) -> cmake::Config {
131        if is_fips_build() {
132            unsafe {
133                env::set_var("GOFLAGS", "-buildvcs=false");
134            }
135            if env::var("GOCACHE").is_err() {
136                unsafe {
137                    env::set_var(
138                        "GOCACHE",
139                        self.out_dir.join(Self::GOCACHE_DIR_NAME).as_os_str(),
140                    );
141                }
142            }
143        }
144
145        let mut cmake_cfg = self.get_cmake_config();
146        if let Some(generator) = optional_env_optional_crate_target("CMAKE_GENERATOR") {
147            set_env("CMAKE_GENERATOR", generator);
148        }
149
150        if OutputLibType::default() == OutputLibType::Dynamic {
151            cmake_cfg.define("BUILD_SHARED_LIBS", "1");
152            if is_fips_build() {
153                // The default flags include `-ffunction-sections` that can result in
154                // dead code elimination dropping functions.
155                cmake_cfg.no_default_flags(true);
156            }
157        } else {
158            cmake_cfg.define("BUILD_SHARED_LIBS", "0");
159        }
160
161        if let Some(prefix) = &self.build_prefix {
162            cmake_cfg.define("BORINGSSL_PREFIX", format!("{prefix}_"));
163            let include_path = self.manifest_dir.join("generated-include");
164            cmake_cfg.define(
165                "BORINGSSL_PREFIX_HEADERS",
166                include_path.display().to_string(),
167            );
168        }
169
170        // Build flags that minimize our crate size.
171        cmake_cfg.define("BUILD_TESTING", "OFF");
172        cmake_cfg.define("BUILD_TOOL", "OFF");
173        cmake_cfg.define("ENABLE_SOURCE_MODIFICATION", "OFF");
174        if cfg!(feature = "ssl") {
175            cmake_cfg.define("BUILD_LIBSSL", "ON");
176        } else {
177            cmake_cfg.define("BUILD_LIBSSL", "OFF");
178        }
179        if is_fips_build() {
180            cmake_cfg.define("FIPS", "1");
181        } else {
182            if is_no_pregenerated_src() {
183                // Go and Perl will be required.
184                cmake_cfg.define("DISABLE_PERL", "OFF");
185                cmake_cfg.define("DISABLE_GO", "OFF");
186            } else {
187                // Build flags that minimize our dependencies.
188                cmake_cfg.define("DISABLE_PERL", "ON");
189                cmake_cfg.define("DISABLE_GO", "ON");
190            }
191            if !should_build_jitter_entropy() {
192                cmake_cfg.define("DISABLE_CPU_JITTER_ENTROPY", "ON");
193            }
194
195            if target_env() == "ohos" {
196                Self::configure_open_harmony(&mut cmake_cfg);
197            }
198        }
199
200        if is_no_asm() {
201            let opt_level = cargo_env("OPT_LEVEL");
202            if opt_level == "0" {
203                cmake_cfg.define("OPENSSL_NO_ASM", "1");
204            } else {
205                panic!("AWS_LC_SYS_NO_ASM only allowed for debug builds!")
206            }
207        }
208
209        Self::configure_sanitizer(&mut cmake_cfg);
210
211        if let Some(cflags) = get_crate_cflags() {
212            let cflags = if is_fips_build() {
213                strip_lto_flags(&cflags)
214            } else {
215                cflags
216            };
217            // Set both the target-specific and base CFLAGS. cmake-rs reads
218            // the base CFLAGS env var directly, so we must override it too.
219            set_env_for_target("CFLAGS", &cflags);
220            set_env("CFLAGS", &cflags);
221        }
222
223        // cmake-rs has logic that strips Optimization/Debug options that are passed via CFLAGS:
224        // https://github.com/rust-lang/cmake-rs/issues/240
225        // This breaks build configurations that generate warnings when optimizations
226        // are disabled.
227        Self::preserve_cflag_optimization_flags(&mut cmake_cfg);
228
229        if target_os() == "windows" {
230            if is_fips_build() {
231                let opt_level = cargo_env("OPT_LEVEL");
232                let build_type = if opt_level.eq("0") || opt_level.eq("1") || opt_level.eq("2") {
233                    "RELWITHDEBINFO"
234                } else {
235                    "RELEASE"
236                };
237                cmake_cfg.define("CMAKE_BUILD_TYPE", build_type);
238            } else if use_prebuilt_nasm() {
239                self.configure_prebuilt_nasm(&mut cmake_cfg);
240            }
241            if target_is_msvc() {
242                let mut msvcrt = String::from_str("MultiThreaded").unwrap();
243                if is_crt_static() {
244                    cmake_cfg.static_crt(true);
245                    // When using static CRT on Windows MSVC, ignore missing PDB file warnings
246                    // The static CRT libraries reference PDB files from Microsoft's build servers
247                    // which are not available.
248                    println!("cargo:rustc-link-arg=/ignore:4099");
249                } else {
250                    msvcrt.push_str("DLL");
251                }
252                cmake_cfg.define("CMAKE_MSVC_RUNTIME_LIBRARY", msvcrt.as_str());
253            }
254        }
255
256        // Allow environment to specify CMake toolchain.
257        if let Some(toolchain) = optional_env_optional_crate_target("CMAKE_TOOLCHAIN_FILE") {
258            set_env_for_target("CMAKE_TOOLCHAIN_FILE", toolchain);
259
260            return cmake_cfg;
261        }
262        // We only consider compiler CFLAGS when no cmake toolchain is set
263        self.apply_universal_build_options(&mut cmake_cfg);
264
265        // See issue: https://github.com/aws/aws-lc-rs/issues/453
266        if target_os() == "windows" {
267            self.configure_windows(&mut cmake_cfg);
268        }
269
270        // If the build environment vendor is Apple
271        #[cfg(target_vendor = "apple")]
272        {
273            if target_arch() == "aarch64" {
274                cmake_cfg.define("CMAKE_OSX_ARCHITECTURES", "arm64");
275                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "arm64");
276            }
277            if target_arch() == "x86_64" {
278                cmake_cfg.define("CMAKE_OSX_ARCHITECTURES", "x86_64");
279                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "x86_64");
280            }
281            if target_os().trim() == "ios" {
282                cmake_cfg.define("CMAKE_SYSTEM_NAME", "iOS");
283                if effective_target().ends_with("-ios-sim") || target_arch() == "x86_64" {
284                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "iphonesimulator");
285                } else {
286                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "iphoneos");
287                }
288                cmake_cfg.define("CMAKE_THREAD_LIBS_INIT", "-lpthread");
289            }
290            if target_os().trim() == "macos" {
291                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Darwin");
292                cmake_cfg.define("CMAKE_OSX_SYSROOT", "macosx");
293            }
294            if target_os().trim() == "tvos" {
295                cmake_cfg.define("CMAKE_SYSTEM_NAME", "tvOS");
296                if effective_target().ends_with("-tvos-sim") || target_arch() == "x86_64" {
297                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "appletvsimulator");
298                } else {
299                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "appletvos");
300                }
301            }
302        }
303
304        if target_os() == "android" {
305            self.configure_android(&mut cmake_cfg);
306        }
307
308        cmake_cfg
309    }

311    fn preserve_cflag_optimization_flags(cmake_cfg: &mut cmake::Config) {
312        if let Some(cflags) = get_crate_cflags() {
313            let split = cflags.split_whitespace();
314            for arg in split {
315                if arg.starts_with("-O") || arg.starts_with("/O") {
316                    emit_warning(format!("Preserving optimization flag: {arg}"));
317                    cmake_cfg.cflag(arg);
318                }
319            }
320        }
321    }

323    #[allow(clippy::unused_self)]
324    fn select_prebuilt_nasm_script(&self) -> PathBuf {
325        let sh_script = self.manifest_dir.join("builder").join("prebuilt-nasm.sh");
326        let bat_script = self.manifest_dir.join("builder").join("prebuilt-nasm.bat");
327
328        // Test .sh first (more universal - works in MSYS2, WSL, native Unix)
329        if test_prebuilt_nasm_script(&sh_script) {
330            emit_warning("Selected prebuilt-nasm.sh (shell script can execute)");
331            sh_script
332        } else if test_prebuilt_nasm_script(&bat_script) {
333            emit_warning(
334                "Selected prebuilt-nasm.bat (batch script can execute, shell script cannot)",
335            );
336            bat_script
337        } else {
338            // Neither script could be verified as executable. This can happen in sandboxed or
339            // restricted build environments where trial execution is blocked. We fall back to
340            // selecting a script based on the host OS, which matches the behavior prior to
341            // execution-based detection. If this fallback is incorrect, the build will fail
342            // when CMake attempts to invoke the selected script.
343            let fallback_script = if cfg!(target_os = "windows") {
344                bat_script
345            } else {
346                sh_script
347            };
348            emit_warning(format!(
349                "WARNING: Neither prebuilt-nasm.sh nor prebuilt-nasm.bat could be verified as \
350                     executable. Falling back to target-based selection: {}. If the build fails \
351                     during assembly, verify that the selected script is appropriate for your \
352                     build environment.",
353                fallback_script.file_name().unwrap().to_str().unwrap()
354            ));
355            fallback_script
356        }
357    }

383    #[allow(clippy::unused_self)]
384    fn configure_windows(&self, cmake_cfg: &mut cmake::Config) {
385        if is_fips_build() {
386            cmake_cfg.generator("Ninja");
387            let env_map = self
388                .collect_vcvarsall_bat()
389                .map_err(|x| panic!("{}", x))
390                .unwrap();
391            for (key, value) in env_map {
392                cmake_cfg.env(key, value);
393            }
394        }
395
396        match (target_env().as_str(), target_arch().as_str()) {
397            ("msvc", "aarch64") => {
398                // If CMAKE_GENERATOR is either not set or not set to "Ninja"
399                let cmake_generator = optional_env("CMAKE_GENERATOR");
400                if cmake_generator.is_none() || cmake_generator.unwrap().to_lowercase() != "ninja" {
401                    // The following is not supported by the Ninja generator
402                    cmake_cfg.generator_toolset(format!(
403                        "ClangCL{}",
404                        if cfg!(target_arch = "x86_64") {
405                            ",host=x64"
406                        } else {
407                            ""
408                        }
409                    ));
410                    cmake_cfg.define("CMAKE_GENERATOR_PLATFORM", "ARM64");
411                }
412                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Windows");
413                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "ARM64");
414            }
415            ("msvc", _) => {
416                // No-op
417            }
418            (_, arch) => {
419                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Windows");
420                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", arch);
421            }
422        }
423
424        // Workaround for win7-windows-gnu targets: the upstream C source gates the
425        // Win7 compat path with `!defined(__MINGW32__)`. CMakeLists.txt already sets
426        // _WIN32_WINNT for MinGW, but we must force AWSLC_WINDOWS_7_COMPAT until the
427        // upstream fix lands: https://github.com/aws/aws-lc/pull/3239
428        if target().contains("-win7-windows-gnu") {
429            cmake_cfg.cflag("-DAWSLC_WINDOWS_7_COMPAT");
430        }
431    }

433    /// Returns the architecture argument for `vcvarsall.bat`.
434    ///
435    /// In a Cargo build-script, `cfg!(target_arch)` is the **host** architecture
436    /// (the machine running the build), while `target_arch()` (backed by
437    /// `CARGO_CFG_TARGET_ARCH`) is the **target** architecture we are compiling for.
438    ///
439    /// `vcvarsall.bat` accepts a single token that encodes both:
440    ///   - native   : `x64`, `arm64`, `x86`
441    ///   - cross    : `<host>_<target>`, e.g. `x64_arm64`
442    fn vcvarsall_arch() -> &'static str {
443        let target = target_arch();
444        match (
445            cfg!(target_arch = "x86_64"),
446            cfg!(target_arch = "aarch64"),
447            target.as_str(),
448        ) {
449            // Host x64
450            (true, _, "x86_64") => "x64",
451            (true, _, "aarch64") => "x64_arm64",
452            (true, _, "x86") => "x64_x86",
453            // Host arm64
454            (_, true, "aarch64") => "arm64",
455            (_, true, "x86_64") => "arm64_x64",
456            (_, true, "x86") => "arm64_x86",
457            // Fallback
458            _ => "x64",
459        }
460    }

462    fn collect_vcvarsall_bat(&self) -> Result<HashMap<String, String>, String> {
463        let mut map: HashMap<String, String> = HashMap::new();
464        let script_path = self.manifest_dir.join("builder").join("printenv.bat");
465        let arch = OsString::from(Self::vcvarsall_arch());
466        let result = execute_command(script_path.as_os_str(), &[arch.as_os_str()]);
467        if !result.status {
468            eprintln!("{}", result.stdout);
469            return Err("Failed to run vcvarsall.bat.".to_owned());
470        }
471        eprintln!("{}", result.stdout);
472        let lines = result.stdout.lines();
473        for line in lines {
474            if let Some((var, val)) = line.split_once('=') {
475                map.insert(var.to_string(), val.to_string());
476            }
477        }
478        Ok(map)
479    }

517    /// Configure the `CMake` build for sanitizer instrumentation.
518    ///
519    /// Resolves the active sanitizer from either the `asan` Cargo feature flag
520    /// (kept for backward compatibility) or the `AWS_LC_SYS_SANITIZER` /
521    /// `AWS_LC_FIPS_SYS_SANITIZER` environment variable (preferred). Supported
522    /// values: `asan`, `msan`, `tsan`.
523    fn configure_sanitizer(cmake_cfg: &mut cmake::Config) {
524        let feature_sanitizer = if cfg!(feature = "asan") {
525            Some("asan")
526        } else {
527            None
528        };
529        let env_sanitizer = sanitizer();
530        let env_sanitizer = env_sanitizer.as_deref();
531
532        let active_sanitizer = match (feature_sanitizer, env_sanitizer) {
533            (Some(f), Some(e)) if f != e => {
534                panic!(
535                    "Conflicting sanitizer configuration: feature flag '{f}' and \
536                     AWS_LC_SYS_SANITIZER='{e}'. Remove one to resolve the conflict."
537                );
538            }
539            (Some(f), _) => Some(f),
540            (_, Some(e)) => Some(e),
541            _ => None,
542        };
543
544        if let Some(san) = active_sanitizer {
545            set_env_for_target("CC", "clang");
546            set_env_for_target("CXX", "clang++");
547
548            match san {
549                "asan" => {
550                    cmake_cfg.define("ASAN", "1");
551                }
552                "msan" => {
553                    // AWS-LC's CMakeLists.txt correctly skips the FIPS
554                    // delocator when MSAN is set, but bcm.c guards the
555                    // integrity-test symbols with `#if !defined(OPENSSL_ASAN)`
556                    // only — it does not check OPENSSL_MSAN. Until the
557                    // upstream C guard is broadened, MSAN + FIPS will produce
558                    // undefined-symbol link errors for BORINGSSL_bcm_text_*.
559                    assert!(
560                        !is_fips_build(),
561                        "MemorySanitizer is not supported for FIPS builds. \
562                         The FIPS integrity check in bcm.c lacks an \
563                         OPENSSL_MSAN guard, causing undefined-symbol link \
564                         errors (BORINGSSL_bcm_text_start/end/hash)."
565                    );
566                    cmake_cfg.define("MSAN", "1");
567                }
568                "tsan" => {
569                    cmake_cfg.define("TSAN", "1");
570                    // AWS-LC's CMakeLists.txt defines BORINGSSL_DISPATCH_TEST for
571                    // non-FIPS, non-release builds. That macro enables non-atomic
572                    // writes to a global BORINGSSL_function_hit[] array from ~15
573                    // dispatch points (C and assembly), which TSAN flags as data
574                    // races. The writes are benign (idempotent store of 1), but
575                    // suppressing every function name is impractical. Force the C
576                    // library to build in Release mode so the macro is never
577                    // defined. This matches AWS-LC's own sanitizer CI, which
578                    // always uses -DCMAKE_BUILD_TYPE=Release.
579                    cmake_cfg.profile("Release");
580                }
581                other => {
582                    panic!("Unsupported sanitizer: '{other}'. Supported values: asan, msan, tsan")
583                }
584            }
585        }
586    }
587
588    fn configure_open_harmony(cmake_cfg: &mut cmake::Config) {
589        let mut cflags = vec!["-Wno-unused-command-line-argument"];
590        let mut asmflags = vec![];
591
592        // If a toolchain is not specified by the environment
593        if optional_env_optional_crate_target("CMAKE_TOOLCHAIN_FILE").is_none() {
594            if let Ok(ndk) = env::var("OHOS_NDK_HOME") {
595                set_env_for_target(
596                    "CMAKE_TOOLCHAIN_FILE",
597                    format!("{ndk}/native/build/cmake/ohos.toolchain.cmake"),
598                );
599            } else if let Ok(sdk) = env::var("OHOS_SDK_NATIVE") {
600                set_env_for_target(
601                    "CMAKE_TOOLCHAIN_FILE",
602                    format!("{sdk}/build/cmake/ohos.toolchain.cmake"),
603                );
604            } else {
605                emit_warning(
606                    "Neither OHOS_NDK_HOME nor OHOS_SDK_NATIVE are set! No toolchain found.",
607                );
608            }
609        }
610
611        match effective_target().as_str() {
612            "aarch64-unknown-linux-ohos" => {
613                cmake_cfg.define("OHOS_ARCH", "arm64-v8a");
614            }
615            "armv7-unknown-linux-ohos" => {
616                const ARM7_FLAGS: [&str; 6] = [
617                    "-march=armv7-a",
618                    "-mfloat-abi=softfp",
619                    "-mtune=generic-armv7-a",
620                    "-mthumb",
621                    "-mfpu=neon",
622                    "-DHAVE_NEON",
623                ];
624                cflags.extend(ARM7_FLAGS);
625                asmflags.extend(ARM7_FLAGS);
626                cmake_cfg.define("OHOS_ARCH", "armeabi-v7a");
627            }
628            "x86_64-unknown-linux-ohos" => {
629                const X86_64_FLAGS: [&str; 3] = ["-msse4.1", "-DHAVE_NEON_X86", "-DHAVE_NEON"];
630                cflags.extend(X86_64_FLAGS);
631                asmflags.extend(X86_64_FLAGS);
632                cmake_cfg.define("OHOS_ARCH", "x86_64");
633            }
634            ohos_target => {
635                emit_warning(format!("Target: {ohos_target} is not support yet!").as_str());
636            }
637        }
638        cmake_cfg
639            .cflag(cflags.join(" ").as_str())
640            .cxxflag(cflags.join(" ").as_str())
641            .asmflag(asmflags.join(" ").as_str());
642    }
643
644    fn build_library(&self) -> PathBuf {
645        self.prepare_cmake_build()
646            .configure_arg("--no-warn-unused-cli")
647            .build()
648    }
649}

651impl crate::Builder for CmakeBuilder {
652    fn check_dependencies(&self) -> Result<(), String> {
653        let mut missing_dependency = false;
654        if target_os() == "windows" && target_arch() == "x86_64" {
655            if is_no_asm() && Some(true) == allow_prebuilt_nasm() {
656                eprintln!(
657                    "Build environment has both `AWS_LC_SYS_PREBUILT_NASM` and `AWS_LC_SYS_NO_ASM` set.\
658                Please remove one of these environment variables.
659                See User Guide: https://aws.github.io/aws-lc-rs/index.html"
660                );
661            }
662            if !is_no_asm() && !test_nasm_command() && !use_prebuilt_nasm() {
663                eprintln!(
664                    "Consider installing NASM or setting `AWS_LC_SYS_PREBUILT_NASM` in the build environment.\
665                See User Guide: https://aws.github.io/aws-lc-rs/index.html"
666                );
667                eprintln!("Missing dependency: nasm");
668                missing_dependency = true;
669            }
670        }
671        if target_os() == "windows"
672            && target_arch() == "aarch64"
673            && target_is_msvc()
674            && !test_clang_cl_command()
675        {
676            eprintln!("Missing dependency: clang-cl");
677            missing_dependency = true;
678        }
679        if let Some(cmake_cmd) = find_cmake_command() {
680            unsafe {
681                env::set_var("CMAKE", cmake_cmd);
682            }
683        } else {
684            eprintln!("Missing dependency: cmake");
685            missing_dependency = true;
686        }
687
688        if missing_dependency {
689            return Err("Required build dependency is missing. Halting build.".to_owned());
690        }
691
692        Ok(())
693    }
694    fn build(&self) -> Result<(), String> {
695        self.build_library();
696
697        println!(
698            "cargo:rustc-link-search=native={}",
699            self.artifact_output_dir().display()
700        );
701
702        println!(
703            "cargo:rustc-link-lib={}={}",
704            self.output_lib_type.rust_link_lib_kind(),
705            Crypto.libname(&self.build_prefix)
706        );
707
708        if cfg!(feature = "ssl") {
709            println!(
710                "cargo:rustc-link-lib={}={}",
711                self.output_lib_type.rust_link_lib_kind(),
712                Ssl.libname(&self.build_prefix)
713            );
714        }
715
716        crate::emit_source_build_metadata(&self.manifest_dir);
717
718        Ok(())
719    }
720
721    fn name(&self) -> &'static str {
722        "CMake"
723    }

727mod tests {
728    use super::strip_lto_flags;
729
730    #[test]
731    fn test_strip_lto_flags() {
732        // RPM-style CFLAGS with -flto=auto and -ffat-lto-objects
733        let input = "-O2 -ftree-vectorize -flto=auto -ffat-lto-objects -fexceptions -g -fPIC";
734        assert_eq!(
735            strip_lto_flags(input),
736            "-O2 -ftree-vectorize -fexceptions -g -fPIC"
737        );
738
739        // Various LTO flag forms
740        assert_eq!(strip_lto_flags("-flto -O2"), "-O2");
741        assert_eq!(strip_lto_flags("-flto=thin -O2"), "-O2");
742        assert_eq!(strip_lto_flags("-flto=full -O2"), "-O2");
743        assert_eq!(strip_lto_flags("-flto=4 -O2"), "-O2");
744        assert_eq!(strip_lto_flags("-fno-fat-lto-objects -O2"), "-O2");
745
746        // No LTO flags - passthrough
747        let no_lto = "-O2 -fPIC -g";
748        assert_eq!(strip_lto_flags(no_lto), no_lto);
749
750        // Empty input
751        assert_eq!(strip_lto_flags(""), "");
752    }
