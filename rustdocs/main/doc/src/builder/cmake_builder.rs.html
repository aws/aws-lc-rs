22/// Strip LTO-related flags from a CFLAGS string. LTO flags interfere with the
23/// FIPS delocator pipeline, which compiles C to assembly (`-S`) and processes it
24/// with a Go tool. The combination of `-S` and `-flto` causes GCC to produce
25/// GIMPLE-annotated output that the delocator cannot handle. Additionally, LTO
26/// object files mixed with hand-written assembly in static libraries can cause
27/// linker errors (e.g. `R_X86_64_32` relocations incompatible with PIE).
28/// See also: aws-lc `CMakeLists.txt` which disables assembly when LTO is active.
29fn strip_lto_flags(cflags: &str) -> String {
30    cflags
31        .split_whitespace()
32        .filter(|flag| !is_lto_flag(flag))
33        .collect::<Vec<_>>()
34        .join(" ")
35}
36
37const fn cmake_bool(value: bool) -> &'static str {
38    if value {
39        "ON"
40    } else {
41        "OFF"
42    }
43}

45const fn cmake_nasm_small_flags(value: bool) -> &'static str {
46    if value {
47        "-DOPENSSL_SMALL=1 -DMY_ASSEMBLER_IS_TOO_OLD_FOR_512AVX=1"
48    } else {
49        ""
50    }
51}

60fn test_prebuilt_nasm_script(script_path: &Path) -> bool {

65fn find_cmake_command() -> Option<OsString> {
66    if let Some(cmake) = optional_env_optional_crate_target("CMAKE") {
67        emit_warning(format!("CMAKE environment variable set: {}", cmake.clone()));
68        if execute_command(cmake.as_ref(), &["--version".as_ref()]).status {
69            Some(cmake.into())
70        } else {
71            None
72        }
73    } else if execute_command("cmake3".as_ref(), &["--version".as_ref()]).status {
74        Some("cmake3".into())
75    } else if execute_command("cmake".as_ref(), &["--version".as_ref()]).status {
76        Some("cmake".into())
77    } else {
78        None
79    }
80}

105    fn apply_universal_build_options<'a>(
106        &self,
107        cmake_cfg: &'a mut cmake::Config,
108    ) -> &'a cmake::Config {
109        // Use the compiler options identified by CcBuilder
110        let cc_builder = CcBuilder::new(
111            self.manifest_dir.clone(),
112            self.out_dir.clone(),
113            self.build_prefix.clone(),
114            self.output_lib_type,
115        );
116        let cc_build = cc::Build::new();
117        let (is_cl_like, build_options) =
118            cc_builder.collect_universal_build_options(&cc_build, true);
119        for option in &build_options {
120            option.apply_cmake(cmake_cfg, is_cl_like);
121        }
122
123        // CMake seeds `CMAKE_C_FLAGS` and `CMAKE_ASM_FLAGS` separately, so the
124        // C-side prefix-map does not cover `.S` sources. In release-style
125        // builds, mirror it with `asmflag()` when the compiler accepts
126        // `-Wa,--debug-prefix-map=...`, and skip paths with spaces because the
127        // flag must survive CMake and the assembler as one bare token.
128        let opt_level = cargo_env("OPT_LEVEL");
129        if !is_cl_like
130            && (target_os() == "linux" || target_os().ends_with("bsd"))
131            && !matches!(opt_level.as_str(), "0" | "1" | "2")
132            && !self.manifest_dir.to_string_lossy().contains(' ')
133        {
134            let path_str = self.manifest_dir.display().to_string();
135            let asm_flag = format!("-Wa,--debug-prefix-map={path_str}=");
136            if cc_build.is_flag_supported(&asm_flag).unwrap_or(false) {
137                cmake_cfg.asmflag(asm_flag);
138            }
139        }
140
141        cmake_cfg
142    }

146    #[allow(clippy::too_many_lines)]
147    fn prepare_cmake_build(&self) -> cmake::Config {
148        if is_fips_build() {
149            unsafe {
150                env::set_var("GOFLAGS", "-buildvcs=false");
151            }
152            if env::var("GOCACHE").is_err() {
153                unsafe {
154                    env::set_var(
155                        "GOCACHE",
156                        self.out_dir.join(Self::GOCACHE_DIR_NAME).as_os_str(),
157                    );
158                }
159            }
160        }
161
162        let mut cmake_cfg = self.get_cmake_config();
163        if let Some(generator) = optional_env_optional_crate_target("CMAKE_GENERATOR") {
164            set_env("CMAKE_GENERATOR", generator);
165        }
166
167        if OutputLibType::default() == OutputLibType::Dynamic {
168            cmake_cfg.define("BUILD_SHARED_LIBS", "1");
169            if is_fips_build() {
170                // The default flags include `-ffunction-sections` that can result in
171                // dead code elimination dropping functions.
172                cmake_cfg.no_default_flags(true);
173            }
174        } else {
175            cmake_cfg.define("BUILD_SHARED_LIBS", "0");
176        }
177
178        if let Some(prefix) = &self.build_prefix {
179            cmake_cfg.define("BORINGSSL_PREFIX", format!("{prefix}_"));
180            let include_path = self.manifest_dir.join("generated-include");
181            cmake_cfg.define(
182                "BORINGSSL_PREFIX_HEADERS",
183                include_path.display().to_string(),
184            );
185        }
186
187        // Build flags that minimize our crate size.
188        cmake_cfg.define("BUILD_TESTING", "OFF");
189        cmake_cfg.define("BUILD_TOOL", "OFF");
190        cmake_cfg.define("ENABLE_SOURCE_MODIFICATION", "OFF");
191        if cfg!(feature = "ssl") {
192            cmake_cfg.define("BUILD_LIBSSL", "ON");
193        } else {
194            cmake_cfg.define("BUILD_LIBSSL", "OFF");
195        }
196        if is_fips_build() {
197            cmake_cfg.define("FIPS", "1");
198        } else {
199            if is_no_pregenerated_src() {
200                // Go and Perl will be required.
201                cmake_cfg.define("DISABLE_PERL", "OFF");
202                cmake_cfg.define("DISABLE_GO", "OFF");
203            } else {
204                // Build flags that minimize our dependencies.
205                cmake_cfg.define("DISABLE_PERL", "ON");
206                cmake_cfg.define("DISABLE_GO", "ON");
207            }
208            if !should_build_jitter_entropy() {
209                cmake_cfg.define("DISABLE_CPU_JITTER_ENTROPY", "ON");
210            }
211
212            if target_env() == "ohos" {
213                Self::configure_open_harmony(&mut cmake_cfg);
214            }
215        }
216
217        if is_no_asm() {
218            let opt_level = cargo_env("OPT_LEVEL");
219            if opt_level == "0" {
220                cmake_cfg.define("OPENSSL_NO_ASM", "1");
221            } else {
222                panic!("AWS_LC_SYS_NO_ASM only allowed for debug builds!")
223            }
224        }
225
226        // Always set these cache entries (ON/OFF, flags/empty): the CMake cache in
227        // OUT_DIR persists across builds, so a stale value could survive a toggle.
228        let small = is_small();
229        let small_value = cmake_bool(small);
230        cmake_cfg.define("OPENSSL_SMALL", small_value);
231        if is_fips_crate() {
232            // Only the pinned FIPS branch needs this; mainline CMakeLists
233            // derives it from OPENSSL_SMALL.
234            cmake_cfg.define("MY_ASSEMBLER_IS_TOO_OLD_FOR_512AVX", small_value);
235        }
236        if target_os() == "windows" && target_arch() == "x86_64" {
237            // Raw NASM cannot include target.h, and add_definitions does not reach it.
238            cmake_cfg.define("CMAKE_ASM_NASM_FLAGS", cmake_nasm_small_flags(small));
239        }
240        if small {
241            emit_warning(
242                "Size-optimized AWS-LC build enabled. Applying OPENSSL_SMALL and disabling AVX-512 assembly.",
243            );
244        }
245
246        Self::configure_sanitizer(&mut cmake_cfg);
247
248        if let Some(cflags) = get_crate_cflags() {
249            let cflags = if is_fips_build() {
250                strip_lto_flags(&cflags)
251            } else {
252                cflags
253            };
254            // Set both the target-specific and base CFLAGS. cmake-rs reads
255            // the base CFLAGS env var directly, so we must override it too.
256            set_env_for_target("CFLAGS", &cflags);
257            set_env("CFLAGS", &cflags);
258        }
259
260        // cmake-rs has logic that strips Optimization/Debug options that are passed via CFLAGS:
261        // https://github.com/rust-lang/cmake-rs/issues/240
262        // This breaks build configurations that generate warnings when optimizations
263        // are disabled.
264        Self::preserve_cflag_optimization_flags(&mut cmake_cfg);
265
266        if target_os() == "windows" {
267            if is_fips_build() {
268                let opt_level = cargo_env("OPT_LEVEL");
269                let build_type = if opt_level.eq("0") || opt_level.eq("1") || opt_level.eq("2") {
270                    "RELWITHDEBINFO"
271                } else {
272                    "RELEASE"
273                };
274                cmake_cfg.define("CMAKE_BUILD_TYPE", build_type);
275            } else if use_prebuilt_nasm() {
276                self.configure_prebuilt_nasm(&mut cmake_cfg);
277            }
278            if target_is_msvc() {
279                let mut msvcrt = String::from_str("MultiThreaded").unwrap();
280                if is_crt_static() {
281                    cmake_cfg.static_crt(true);
282                    // When using static CRT on Windows MSVC, ignore missing PDB file warnings
283                    // The static CRT libraries reference PDB files from Microsoft's build servers
284                    // which are not available.
285                    println!("cargo:rustc-link-arg=/ignore:4099");
286                } else {
287                    msvcrt.push_str("DLL");
288                }
289                cmake_cfg.define("CMAKE_MSVC_RUNTIME_LIBRARY", msvcrt.as_str());
290            }
291        }
292
293        // Allow environment to specify CMake toolchain.
294        if let Some(toolchain) = optional_env_optional_crate_target("CMAKE_TOOLCHAIN_FILE") {
295            set_env_for_target("CMAKE_TOOLCHAIN_FILE", toolchain);
296
297            return cmake_cfg;
298        }
299        // We only consider compiler CFLAGS when no cmake toolchain is set
300        self.apply_universal_build_options(&mut cmake_cfg);
301
302        // See issue: https://github.com/aws/aws-lc-rs/issues/453
303        if target_os() == "windows" {
304            self.configure_windows(&mut cmake_cfg);
305        }
306
307        // If the build environment vendor is Apple
308        #[cfg(target_vendor = "apple")]
309        {
310            if target_arch() == "aarch64" {
311                cmake_cfg.define("CMAKE_OSX_ARCHITECTURES", "arm64");
312                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "arm64");
313            }
314            if target_arch() == "x86_64" {
315                cmake_cfg.define("CMAKE_OSX_ARCHITECTURES", "x86_64");
316                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "x86_64");
317            }
318            if target_os().trim() == "ios" {
319                cmake_cfg.define("CMAKE_SYSTEM_NAME", "iOS");
320                if effective_target().ends_with("-ios-sim") || target_arch() == "x86_64" {
321                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "iphonesimulator");
322                } else {
323                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "iphoneos");
324                }
325                cmake_cfg.define("CMAKE_THREAD_LIBS_INIT", "-lpthread");
326            }
327            if target_os().trim() == "macos" {
328                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Darwin");
329                cmake_cfg.define("CMAKE_OSX_SYSROOT", "macosx");
330            }
331            if target_os().trim() == "tvos" {
332                cmake_cfg.define("CMAKE_SYSTEM_NAME", "tvOS");
333                if effective_target().ends_with("-tvos-sim") || target_arch() == "x86_64" {
334                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "appletvsimulator");
335                } else {
336                    cmake_cfg.define("CMAKE_OSX_SYSROOT", "appletvos");
337                }
338            }
339        }
340
341        if target_os() == "android" {
342            self.configure_android(&mut cmake_cfg);
343        }
344
345        cmake_cfg
346    }

348    fn preserve_cflag_optimization_flags(cmake_cfg: &mut cmake::Config) {
349        if let Some(cflags) = get_crate_cflags() {
350            let split = cflags.split_whitespace();
351            for arg in split {
352                if arg.starts_with("-O") || arg.starts_with("/O") {
353                    emit_warning(format!("Preserving optimization flag: {arg}"));
354                    cmake_cfg.cflag(arg);
355                }
356            }
357        }
358    }

360    #[allow(clippy::unused_self)]
361    fn select_prebuilt_nasm_script(&self) -> PathBuf {
362        let sh_script = self.manifest_dir.join("builder").join("prebuilt-nasm.sh");
363        let bat_script = self.manifest_dir.join("builder").join("prebuilt-nasm.bat");
364
365        // Test .sh first (more universal - works in MSYS2, WSL, native Unix)
366        if test_prebuilt_nasm_script(&sh_script) {
367            emit_warning("Selected prebuilt-nasm.sh (shell script can execute)");
368            sh_script
369        } else if test_prebuilt_nasm_script(&bat_script) {
370            emit_warning(
371                "Selected prebuilt-nasm.bat (batch script can execute, shell script cannot)",
372            );
373            bat_script
374        } else {
375            // Neither script could be verified as executable. This can happen in sandboxed or
376            // restricted build environments where trial execution is blocked. We fall back to
377            // selecting a script based on the host OS, which matches the behavior prior to
378            // execution-based detection. If this fallback is incorrect, the build will fail
379            // when CMake attempts to invoke the selected script.
380            let fallback_script = if cfg!(target_os = "windows") {
381                bat_script
382            } else {
383                sh_script
384            };
385            emit_warning(format!(
386                "WARNING: Neither prebuilt-nasm.sh nor prebuilt-nasm.bat could be verified as \
387                     executable. Falling back to target-based selection: {}. If the build fails \
388                     during assembly, verify that the selected script is appropriate for your \
389                     build environment.",
390                fallback_script.file_name().unwrap().to_str().unwrap()
391            ));
392            fallback_script
393        }
394    }

420    #[allow(clippy::unused_self)]
421    fn configure_windows(&self, cmake_cfg: &mut cmake::Config) {
422        if is_fips_build() {
423            cmake_cfg.generator("Ninja");
424            let env_map = self
425                .collect_vcvarsall_bat()
426                .map_err(|x| panic!("{}", x))
427                .unwrap();
428            for (key, value) in env_map {
429                cmake_cfg.env(key, value);
430            }
431        }
432
433        match (target_env().as_str(), target_arch().as_str()) {
434            ("msvc", "aarch64") => {
435                // If CMAKE_GENERATOR is either not set or not set to "Ninja"
436                let cmake_generator = optional_env("CMAKE_GENERATOR");
437                if cmake_generator.is_none() || cmake_generator.unwrap().to_lowercase() != "ninja" {
438                    // The following is not supported by the Ninja generator
439                    cmake_cfg.generator_toolset(format!(
440                        "ClangCL{}",
441                        if cfg!(target_arch = "x86_64") {
442                            ",host=x64"
443                        } else {
444                            ""
445                        }
446                    ));
447                    cmake_cfg.define("CMAKE_GENERATOR_PLATFORM", "ARM64");
448                }
449                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Windows");
450                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", "ARM64");
451            }
452            ("msvc", _) => {
453                // No-op
454            }
455            (_, arch) => {
456                cmake_cfg.define("CMAKE_SYSTEM_NAME", "Windows");
457                cmake_cfg.define("CMAKE_SYSTEM_PROCESSOR", arch);
458            }
459        }
460
461        // Workaround for win7-windows-gnu targets: the upstream C source gates the
462        // Win7 compat path with `!defined(__MINGW32__)`. CMakeLists.txt already sets
463        // _WIN32_WINNT for MinGW, but we must force AWSLC_WINDOWS_7_COMPAT until the
464        // upstream fix lands: https://github.com/aws/aws-lc/pull/3239
465        if target().contains("-win7-windows-gnu") {
466            cmake_cfg.cflag("-DAWSLC_WINDOWS_7_COMPAT");
467        }
468    }

470    /// Returns the architecture argument for `vcvarsall.bat`.
471    ///
472    /// In a Cargo build-script, `cfg!(target_arch)` is the **host** architecture
473    /// (the machine running the build), while `target_arch()` (backed by
474    /// `CARGO_CFG_TARGET_ARCH`) is the **target** architecture we are compiling for.
475    ///
476    /// `vcvarsall.bat` accepts a single token that encodes both:
477    ///   - native   : `x64`, `arm64`, `x86`
478    ///   - cross    : `<host>_<target>`, e.g. `x64_arm64`
479    fn vcvarsall_arch() -> &'static str {
480        let target = target_arch();
481        match (
482            cfg!(target_arch = "x86_64"),
483            cfg!(target_arch = "aarch64"),
484            target.as_str(),
485        ) {
486            // Host x64
487            (true, _, "x86_64") => "x64",
488            (true, _, "aarch64") => "x64_arm64",
489            (true, _, "x86") => "x64_x86",
490            // Host arm64
491            (_, true, "aarch64") => "arm64",
492            (_, true, "x86_64") => "arm64_x64",
493            (_, true, "x86") => "arm64_x86",
494            // Fallback
495            _ => "x64",
496        }
497    }

499    fn collect_vcvarsall_bat(&self) -> Result<HashMap<String, String>, String> {
500        let mut map: HashMap<String, String> = HashMap::new();
501        let script_path = self.manifest_dir.join("builder").join("printenv.bat");
502        let arch = OsString::from(Self::vcvarsall_arch());
503        let result = execute_command(script_path.as_os_str(), &[arch.as_os_str()]);
504        if !result.status {
505            eprintln!("{}", result.stdout);
506            return Err("Failed to run vcvarsall.bat.".to_owned());
507        }
508        eprintln!("{}", result.stdout);
509        let lines = result.stdout.lines();
510        for line in lines {
511            if let Some((var, val)) = line.split_once('=') {
512                map.insert(var.to_string(), val.to_string());
513            }
514        }
515        Ok(map)
516    }

521        emit_warning("!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!");

523            emit_warning(

539            "CMAKE_ASM_NASM_COMPILE_OPTIONS_MSVC_RUNTIME_LIBRARY_MultiThreaded",
540            "",
541        );
542        cmake_cfg.define(
543            "CMAKE_ASM_NASM_COMPILE_OPTIONS_MSVC_RUNTIME_LIBRARY_MultiThreadedDLL",
544            "",
545        );
546        cmake_cfg.define(
547            "CMAKE_ASM_NASM_COMPILE_OPTIONS_MSVC_RUNTIME_LIBRARY_MultiThreadedDebug",

549        );

556            "",
557        );

560    /// Configure the `CMake` build for sanitizer instrumentation.
561    ///
562    /// Resolves the active sanitizer from either the `asan` Cargo feature flag
563    /// (kept for backward compatibility) or the `AWS_LC_SYS_SANITIZER` /
564    /// `AWS_LC_FIPS_SYS_SANITIZER` environment variable (preferred). Supported
565    /// values: `asan`, `msan`, `tsan`.
566    fn configure_sanitizer(cmake_cfg: &mut cmake::Config) {
567        let feature_sanitizer = if cfg!(feature = "asan") {
568            Some("asan")
569        } else {
570            None
571        };
572        let env_sanitizer = sanitizer();
573        let env_sanitizer = env_sanitizer.as_deref();
574
575        let active_sanitizer = match (feature_sanitizer, env_sanitizer) {
576            (Some(f), Some(e)) if f != e => {
577                panic!(
578                    "Conflicting sanitizer configuration: feature flag '{f}' and \
579                     AWS_LC_SYS_SANITIZER='{e}'. Remove one to resolve the conflict."
580                );
581            }
582            (Some(f), _) => Some(f),
583            (_, Some(e)) => Some(e),
584            _ => None,
585        };
586
587        if let Some(san) = active_sanitizer {
588            set_env_for_target("CC", "clang");
589            set_env_for_target("CXX", "clang++");
590
591            match san {
592                "asan" => {
593                    cmake_cfg.define("ASAN", "1");
594                }
595                "msan" => {
596                    // AWS-LC's CMakeLists.txt correctly skips the FIPS
597                    // delocator when MSAN is set, but bcm.c guards the
598                    // integrity-test symbols with `#if !defined(OPENSSL_ASAN)`
599                    // only — it does not check OPENSSL_MSAN. Until the
600                    // upstream C guard is broadened, MSAN + FIPS will produce
601                    // undefined-symbol link errors for BORINGSSL_bcm_text_*.
602                    assert!(
603                        !is_fips_build(),
604                        "MemorySanitizer is not supported for FIPS builds. \
605                         The FIPS integrity check in bcm.c lacks an \
606                         OPENSSL_MSAN guard, causing undefined-symbol link \
607                         errors (BORINGSSL_bcm_text_start/end/hash)."
608                    );
609                    cmake_cfg.define("MSAN", "1");
610                }
611                "tsan" => {
612                    cmake_cfg.define("TSAN", "1");
613                    // AWS-LC's CMakeLists.txt defines BORINGSSL_DISPATCH_TEST for
614                    // non-FIPS, non-release builds. That macro enables non-atomic
615                    // writes to a global BORINGSSL_function_hit[] array from ~15
616                    // dispatch points (C and assembly), which TSAN flags as data
617                    // races. The writes are benign (idempotent store of 1), but
618                    // suppressing every function name is impractical. Force the C
619                    // library to build in Release mode so the macro is never
620                    // defined. This matches AWS-LC's own sanitizer CI, which
621                    // always uses -DCMAKE_BUILD_TYPE=Release.
622                    cmake_cfg.profile("Release");
623                }
624                other => {
625                    panic!("Unsupported sanitizer: '{other}'. Supported values: asan, msan, tsan")
626                }
627            }
628        }
629    }
630
631    fn configure_open_harmony(cmake_cfg: &mut cmake::Config) {
632        let mut cflags = vec!["-Wno-unused-command-line-argument"];
633        let mut asmflags = vec![];
634
635        // If a toolchain is not specified by the environment
636        if optional_env_optional_crate_target("CMAKE_TOOLCHAIN_FILE").is_none() {
637            if let Ok(ndk) = env::var("OHOS_NDK_HOME") {
638                set_env_for_target(
639                    "CMAKE_TOOLCHAIN_FILE",
640                    format!("{ndk}/native/build/cmake/ohos.toolchain.cmake"),
641                );
642            } else if let Ok(sdk) = env::var("OHOS_SDK_NATIVE") {
643                set_env_for_target(
644                    "CMAKE_TOOLCHAIN_FILE",
645                    format!("{sdk}/build/cmake/ohos.toolchain.cmake"),
646                );
647            } else {
648                emit_warning(
649                    "Neither OHOS_NDK_HOME nor OHOS_SDK_NATIVE are set! No toolchain found.",
650                );
651            }
652        }
653
654        match effective_target().as_str() {
655            "aarch64-unknown-linux-ohos" => {
656                cmake_cfg.define("OHOS_ARCH", "arm64-v8a");
657            }
658            "armv7-unknown-linux-ohos" => {
659                const ARM7_FLAGS: [&str; 6] = [
660                    "-march=armv7-a",
661                    "-mfloat-abi=softfp",
662                    "-mtune=generic-armv7-a",
663                    "-mthumb",
664                    "-mfpu=neon",
665                    "-DHAVE_NEON",
666                ];
667                cflags.extend(ARM7_FLAGS);
668                asmflags.extend(ARM7_FLAGS);
669                cmake_cfg.define("OHOS_ARCH", "armeabi-v7a");
670            }
671            "x86_64-unknown-linux-ohos" => {
672                const X86_64_FLAGS: [&str; 3] = ["-msse4.1", "-DHAVE_NEON_X86", "-DHAVE_NEON"];
673                cflags.extend(X86_64_FLAGS);
674                asmflags.extend(X86_64_FLAGS);
675                cmake_cfg.define("OHOS_ARCH", "x86_64");
676            }
677            ohos_target => {
678                emit_warning(format!("Target: {ohos_target} is not support yet!").as_str());
679            }
680        }
681        cmake_cfg
682            .cflag(cflags.join(" ").as_str())
683            .cxxflag(cflags.join(" ").as_str())
684            .asmflag(asmflags.join(" ").as_str());
685    }
686
687    fn build_library(&self) -> PathBuf {
688        self.prepare_cmake_build()
689            .configure_arg("--no-warn-unused-cli")
690            .build()
691    }

694impl crate::Builder for CmakeBuilder {
695    fn check_dependencies(&self) -> Result<(), String> {
696        let mut missing_dependency = false;
697        if target_os() == "windows" && target_arch() == "x86_64" {
698            if is_no_asm() && Some(true) == allow_prebuilt_nasm() {
699                eprintln!(
700                    "Build environment has both `AWS_LC_SYS_PREBUILT_NASM` and `AWS_LC_SYS_NO_ASM` set.\
701                Please remove one of these environment variables.
702                See User Guide: https://aws.github.io/aws-lc-rs/index.html"
703                );
704            }
705            if !is_no_asm() && !test_nasm_command() && !use_prebuilt_nasm() {
706                eprintln!(
707                    "Consider installing NASM or setting `AWS_LC_SYS_PREBUILT_NASM` in the build environment.\
708                See User Guide: https://aws.github.io/aws-lc-rs/index.html"
709                );
710                eprintln!("Missing dependency: nasm");
711                missing_dependency = true;
712            }
713        }
714        if target_os() == "windows"
715            && target_arch() == "aarch64"
716            && target_is_msvc()
717            && !test_clang_cl_command()
718        {
719            eprintln!("Missing dependency: clang-cl");
720            missing_dependency = true;
721        }
722        if let Some(cmake_cmd) = find_cmake_command() {
723            unsafe {
724                env::set_var("CMAKE", cmake_cmd);
725            }
726        } else {
727            eprintln!("Missing dependency: cmake");
728            missing_dependency = true;
729        }
730
731        if missing_dependency {
732            return Err("Required build dependency is missing. Halting build.".to_owned());
733        }
734
735        Ok(())
736    }
737    fn build(&self) -> Result<(), String> {
738        self.build_library();
739
740        crate::emit_source_library_metadata(
741            &self.artifact_output_dir(),
742            self.output_lib_type,
743            &self.build_prefix,
744        );
745
746        println!(
747            "cargo:rustc-link-search=native={}",
748            self.artifact_output_dir().display()
749        );
750
751        println!(
752            "cargo:rustc-link-lib={}={}",
753            self.output_lib_type.rust_link_lib_kind(),
754            Crypto.libname(&self.build_prefix)
755        );
756
757        if cfg!(feature = "ssl") {
758            println!(
759                "cargo:rustc-link-lib={}={}",
760                self.output_lib_type.rust_link_lib_kind(),
761                Ssl.libname(&self.build_prefix)
762            );
763        }
764
765        crate::emit_source_build_metadata(&self.manifest_dir, &self.build_prefix);
766
767        Ok(())
768    }
769
770    fn name(&self) -> &'static str {
771        "CMake"
772    }

794    #[test]
795    fn test_strip_lto_flags() {
796        // RPM-style CFLAGS with -flto=auto and -ffat-lto-objects
797        let input = "-O2 -ftree-vectorize -flto=auto -ffat-lto-objects -fexceptions -g -fPIC";
798        assert_eq!(
799            strip_lto_flags(input),
800            "-O2 -ftree-vectorize -fexceptions -g -fPIC"
801        );
802
803        // Various LTO flag forms
804        assert_eq!(strip_lto_flags("-flto -O2"), "-O2");
805        assert_eq!(strip_lto_flags("-flto=thin -O2"), "-O2");
806        assert_eq!(strip_lto_flags("-flto=full -O2"), "-O2");
807        assert_eq!(strip_lto_flags("-flto=4 -O2"), "-O2");
808        assert_eq!(strip_lto_flags("-fno-fat-lto-objects -O2"), "-O2");
809
810        // No LTO flags - passthrough
811        let no_lto = "-O2 -fPIC -g";
812        assert_eq!(strip_lto_flags(no_lto), no_lto);
813
814        // Empty input
815        assert_eq!(strip_lto_flags(""), "");
816    }
